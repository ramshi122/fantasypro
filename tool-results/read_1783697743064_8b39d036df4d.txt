   167→   167→function randomKeySegment(): string {
   168→   168→  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
   169→   169→  let out = "";
   170→   170→  for (let i = 0; i < 4; i++) {

   172→   172→  }

   174→   174→}

   180→   180→export function getExpiryForType(type: KeyType): number | null {
   181→   181→  const now = Date.now();
   182→   182→  const day = 86400000;
   183→   183→  const hour = 3600000;
   184→   184→  switch (type) {
   185→   185→    case "trial": return now + 1 * day;        // 1 day free
   186→   186→    case "match_basic": return now + 6 * hour; // 1 match = 6 hours
   187→   187→    case "match_full": return now + 1 * day;   // 24 hour active
   188→   188→    case "weekly": return now + 7 * day;       // 7 days
   189→   189→    case "monthly": return now + 30 * day;     // 30 days (Pro ₹299)
   190→   190→    case "elite": return now + 90 * day;       // 90 days (Elite ₹999)
   191→   191→    case "lifetime": return null;              // forever
   192→   192→  }
   193→   193→}
