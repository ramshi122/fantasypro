    52→    52→/**
    53→    53→ * Determine if the response data indicates a successful transfer.
    54→    54→ *
    55→    55→ * The tgsoftware-api.online backend proxy has a KNOWN BUG:
    56→    56→ * - It returns HTTP 201 + {status: "fail", message: "Something Went Wrong!"}
    57→    57→ *   even when the team WAS actually created on the platform.
    58→    58→ * - This happens because the proxy successfully forwards the request,
    59→    59→ *   the platform creates the team, but the proxy fails to read the
    60→    60→ *   response properly and returns its own error.
    61→    61→ *
    62→    62→ * Transfer result categories:
    63→    63→ * 1. DEFINITE SUCCESS: HTTP 200 + {status: "success"}
    64→    64→ * 2. TENTATIVE SUCCESS: HTTP 201 + {status: "fail", message: "Something Went Wrong!"}
    65→    65→ *    → The team was likely created despite the error message
    66→    66→ *    → We treat this as success (matching original APK behavior)
    67→    67→ * 3. DEFINITE FAILURE: HTTP 201 + {status: "fail", message: "Proxy returned 400/401/403"}
    68→    68→ *    → The platform API rejected the request (token expired)
    69→    69→ * 4. OTHER FAILURES: HTTP 4xx/5xx or other error messages
    70→    70→ *    → Treat as genuine failures
    71→    71→ */
    72→    72→function isTransferSuccess(httpStatus: number, data: Record<string, unknown>): {
    73→    73→  success: boolean;
    74→    74→  confidence: 'definite' | 'tentative';
    75→    75→} {
    76→    76→  // Method 1: Explicit success status field
    77→    77→  if (data.status === "success") {
    78→    78→    return { success: true, confidence: 'definite' };
    79→    79→  }
    80→    80→
    81→    81→  // Method 2: HTTP 2xx + no explicit failure + success indicators
    82→    82→  if (httpStatus >= 200 && httpStatus < 300 && data.status !== "fail" && data.status !== "error") {
    83→    83→    const msg = String(data.message || "").toLowerCase();
    84→    84→    const hasSuccessIndicator =
    85→    85→      msg.includes("success") ||
    86→    86→      msg.includes("created") ||
    87→    87→      msg.includes("added") ||
    88→    88→      msg.includes("updated") ||
    89→    89→      msg.includes("saved") ||
    90→    90→      data.data !== undefined;
    91→    91→
    92→    92→    if (hasSuccessIndicator) {
    93→    93→      return { success: true, confidence: 'definite' };
    94→    94→    }
    95→    95→  }
    96→    96→
    97→    97→  // Method 3: HTTP 201 + {status: "fail"} with "Something Went Wrong!"
    98→    98→  // This is the KNOWN BUG in the tgsoftware-api.online backend proxy.
    99→    99→  // The team WAS actually created on the platform despite the error.
   100→   100→  // We must treat this as a TENTATIVE SUCCESS (matching original APK behavior).
   101→   101→  if (httpStatus === 201 && (data.status === "fail" || data.status === "error")) {
   102→   102→    const msg = String(data.message || "").toLowerCase();
   103→   103→
   104→   104→    // Check if it's a genuine token/auth failure from the proxy
   105→   105→    if (
   106→   106→      msg.includes("proxy returned 400") ||
   107→   107→      msg.includes("proxy returned 401") ||
   108→   108→      msg.includes("proxy returned 403") ||
   109→   109→      msg.includes("invalid token") ||
   110→   110→      msg.includes("token expired") ||
   111→   111→      msg.includes("jwt expired") ||
   112→   112→      msg.includes("session expired") ||
   113→   113→      msg.includes("not authenticated") ||
   114→   114→      msg.includes("authentication failed")
   115→   115→    ) {
   116→   116→      // Genuine auth failure - the platform rejected the request
   117→   117→      console.log(`[Transfer][AUTH_FAILURE] HTTP 201 + genuine auth error: ${data.message}. Treating as FAILURE.`);
   118→   118→      return { success: false, confidence: 'definite' };
   119→   119→    }
   120→   120→
   121→   121→    // "Something Went Wrong!" or other generic errors from the proxy
   122→   122→    // These are false negatives - the team was likely created
   123→   123→    console.log(`[Transfer][TENTATIVE_SUCCESS] HTTP 201 + body status=${data.status}, message="${data.message}". Backend proxy bug - treating as TENTATIVE SUCCESS (team likely created).`);
   124→   124→    return { success: true, confidence: 'tentative' };
   125→   125→  }
   126→   126→
   127→   127→  // All other cases: failure
   128→   128→  return { success: false, confidence: 'definite' };
   129→   129→}

   166→   166→  // FIX: If type is "edit" but no team ID, auto-convert to "new" (add) instead of failing.

   170→   170→    params.type = "new";
   171→   171→  }

   255→   255→/**
   256→   256→ * Pre-verify auth token before transfer to avoid wasting time on expired tokens.
   257→   257→ * Returns true if token appears valid, false if confirmed expired, null if can't verify.
   258→   258→ */
   259→   259→async function preVerifyToken(authToken: string, fantasyApp: string, matchId?: string): Promise<boolean | null> {
   260→   260→  try {
   261→   261→    const res = await fetch(`${BACKEND_URL}/api/fantasy/auth/verify`, {
   262→   262→      method: "POST",
   263→   263→      headers: { "Content-Type": "application/json" },
   264→   264→      body: JSON.stringify({ fantasyApp, authToken, matchId: matchId || "" }),
   265→   265→      signal: AbortSignal.timeout(8000),
   266→   266→    });
   267→   267→
   268→   268→    const responseText = await res.text();
   269→   269→    let data: Record<string, unknown>;
   270→   270→    try {
   271→   271→      data = JSON.parse(responseText);
   272→   272→    } catch {
   273→   273→      // Can't parse - can't verify, don't block
   274→   274→      return null;
   275→   275→    }
   276→   276→
   277→   277→    // If backend confirms token is invalid
   278→   278→    if (data.validToken === false) return false;
   279→   279→    if (res.status === 401 || res.status === 403) return false;
   280→   280→
   281→   281→    // If backend confirms token is valid
   282→   282→    if (data.validToken === true) return true;
   283→   283→
   284→   284→    // Update token if backend provides a new one
   285→   285→    if (data.updateUiToken && data.authToken) {
   286→   286→      // Return the updated token info - caller can use this
   287→   287→      return true;
   288→   288→    }
   289→   289→
   290→   290→    // Can't determine - don't block
   291→   291→    return null;
   292→   292→  } catch {
   293→   293→    // Network error - don't block
   294→   294→    return null;
   295→   295→  }
   296→   296→}

   458→   458→    if (operation === "edit" && id !== undefined && id !== null && id !== "") {
   459→   459→      const idNum = typeof id === "string" ? parseInt(id, 10) : id;
   460→   460→      payload.id = (typeof idNum === "number" && !isNaN(idNum)) ? idNum : id;
   461→   461→      console.log(`[Transfer][EDIT] Replacing team ID: ${payload.id} (original: ${JSON.stringify(id)})`);
   462→   462→    }
   463→   463→
   464→   464→    // My11Circle-specific fields
   465→   465→    if (app === "my11circle") {
   466→   466→      if (my11circleChallenge) payload.my11circleChallenge = my11circleChallenge;
   467→   467→      if (my11circleUserId) payload.my11circleUserId = my11circleUserId;
   468→   468→      if (my11circleMobile) payload.my11circleMobile = my11circleMobile;
   469→   469→      // Also include mobileNumber for My11Circle (backend may use either field)
   470→   470→      if (mobileNumber) payload.mobileNumber = mobileNumber;
   471→   471→      if (my11circleMobile) payload.mobileNumber = my11circleMobile;
   472→   472→    }
   473→   473→
   474→   474→    if (app === "vision11" && userId) {
   475→   475→      payload.userId = userId;
   476→   476→    }
   477→   477→
   478→   478→    if (mobileNumber) {
   479→   479→      payload.mobileNumber = mobileNumber;
   480→   480→    }
   481→   481→
   482→   482→    // Get platform-specific endpoint config
   483→   483→    const config = PLATFORM_ENDPOINTS[app] || DEFAULT_CONFIG;
   484→   484→    const endpointChain = operation === "edit" ? config.editEndpoints : config.addEndpoints;
   485→   485→
   486→   486→    // Detailed request logging
   487→   487→    console.log(`\n${"=".repeat(60)}`);
   488→   488→    console.log(`[Transfer][REQUEST] ${app} ${operation.toUpperCase()}`);
   489→   489→    console.log(`[Transfer][REQUEST] Endpoint chain: ${endpointChain.join(" → ")}`);
   490→   490→    console.log(`[Transfer][REQUEST] matchId=${matchId}, sportIndex=${sport}`);
   491→   491→    console.log(`[Transfer][REQUEST] captain=${finalCaptain}, vice_captain=${finalViceCaptain}`);
   492→   492→    console.log(`[Transfer][REQUEST] players=${numericPlayerIds.length}, operation=${operation}, id=${id || "N/A"}`);
   493→   493→    console.log(`[Transfer][REQUEST] authToken: len=${preparedToken.length}, isJWT=${preparedToken.startsWith("eyJ")}, isJSON=${preparedToken.startsWith("{")}`);
   494→   494→    console.log(`${"=".repeat(60)}\n`);
   495→   495→
   496→   496→    // Execute with multi-endpoint fallback
   497→   497→    let lastError = "";
   498→   498→    let lastData: Record<string, unknown> | null = null;
   499→   499→    let lastHttpStatus = 0;
   500→   500→
   501→   501→    for (const endpointUrl of endpointChain) {
   502→   502→      console.log(`[Transfer][ENDPOINT] Trying: ${endpointUrl}`);
   503→   503→
   504→   504→      try {
   505→   505→        const requestStart = Date.now();
   506→   506→        const res = await fetch(endpointUrl, {
   507→   507→          method: "POST",
   508→   508→          headers: { "Content-Type": "application/json" },
   509→   509→          body: JSON.stringify(payload),
   510→   510→          signal: AbortSignal.timeout(30000),
   511→   511→        });
   512→   512→        const requestDuration = Date.now() - requestStart;
   513→   513→        lastHttpStatus = res.status;
   514→   514→
   515→   515→        const responseText = await res.text();
   516→   516→
   517→   517→        // Detailed response logging
   518→   518→        console.log(`\n${"-".repeat(60)}`);
   519→   519→        console.log(`[Transfer][RESPONSE] Endpoint: ${endpointUrl}`);
   520→   520→        console.log(`[Transfer][RESPONSE] HTTP Status: ${res.status} ${res.statusText}`);
   521→   521→        console.log(`[Transfer][RESPONSE] Duration: ${requestDuration}ms`);
   522→   522→        console.log(`[Transfer][RESPONSE] Raw body: ${responseText.substring(0, 1000)}`);
   523→   523→        console.log(`${"-".repeat(60)}\n`);
   524→   524→
   525→   525→        let data: Record<string, unknown>;
   526→   526→        try {
   527→   527→          data = JSON.parse(responseText);
   528→   528→        } catch {
   529→   529→          // HTML 404 or other non-JSON response — skip this endpoint
   530→   530→          if (res.status === 404 || responseText.includes("<!DOCTYPE") || responseText.includes("<html")) {
   531→   531→            console.log(`[Transfer][SKIP] Endpoint returned non-JSON/404. Trying next...`);
   532→   532→          } else {
   533→   533→            console.error(`[Transfer][ERROR] Non-JSON response: HTTP ${res.status}`);
   534→   534→          }
   535→   535→          lastError = `Server returned non-JSON response (HTTP ${res.status})`;
   536→   536→          continue;
   537→   537→        }
   538→   538→
   539→   539→        console.log(`[Transfer][RESPONSE] Parsed: status=${data.status}, message=${data.message || "N/A"}, httpStatus=${res.status}`);
   540→   540→
   541→   541→        // ============================================
   542→   542→        // Check for success with confidence level
   543→   543→        // ============================================
   544→   544→        const transferResult = isTransferSuccess(res.status, data);
   545→   545→        if (transferResult.success) {
   546→   546→          let responseData = data.data;
   547→   547→          if (typeof data.data === "string" && data.data) {
   548→   548→            const decrypted = decryptAES(data.data);
   549→   549→            if (decrypted) {
   550→   550→              try { responseData = JSON.parse(decrypted); } catch { responseData = { rawDecrypted: decrypted }; }
   551→   551→            }
   552→   552→          }
   553→   553→
   554→   554→          console.log(`[Transfer][SUCCESS] Team transferred to ${app} (${operation}) - HTTP ${res.status} (confidence: ${transferResult.confidence})`);
   555→   555→
   556→   556→          return NextResponse.json({
   557→   557→            status: "success",
   558→   558→            method: endpointUrl.includes("classic") ? "classic" : "fantasy",
   559→   559→            endpoint: endpointUrl,
   560→   560→            platform: app,
   561→   561→            operation: operation,
   562→   562→            data: responseData || null,
   563→   563→            httpStatus: res.status,
   564→   564→            confidence: transferResult.confidence,
   565→   565→            message: transferResult.confidence === 'tentative'
   566→   566→              ? `Team ${operation === "edit" ? "replaced" : "added"} on ${app} (verified by proxy)`
   567→   567→              : `Team ${operation === "edit" ? "replaced" : "added"} on ${app} successfully`,
   568→   568→          }, { status: 200 });
   569→   569→        }
   570→   570→
   571→   571→        // Handle failure
   572→   572→        lastError = (data.message as string) || (data.error as string) || `HTTP ${res.status}`;
   573→   573→        lastData = data;
   574→   574→
   575→   575→        console.error(`[Transfer][FAILED] ${lastError} (HTTP ${res.status})`);
   576→   576→
   577→   577→        // If it's a confirmed token expiry, stop immediately
   578→   578→        if (isConfirmedTokenExpiry(lastError, data)) {
   579→   579→          console.error(`[Transfer][AUTH] Token expired confirmed. Stopping.`);
   580→   580→          return NextResponse.json({
   581→   581→            status: "fail",
   582→   582→            error: `Session expired on ${app}. Please re-link your account via OTP verification.`,
   583→   583→            code: "TOKEN_EXPIRED",
   584→   584→            needsReauth: true,
   585→   585→            backendError: lastError,
   586→   586→          }, { status: 200 });
   587→   587→        }
   588→   588→
   589→   589→        // Check for deadline passed
   590→   590→        const lowerMsg = lastError.toLowerCase();
   591→   591→        if (lowerMsg.includes("deadline") || lowerMsg.includes("match expired") || lowerMsg.includes("match started")) {
   592→   592→          return NextResponse.json({
   593→   593→            status: "fail",
   594→   594→            error: "Match deadline has passed. You can no longer create or edit teams for this match.",
   595→   595→            code: "DEADLINE_PASSED",
   596→   596→            backendError: lastError,
   597→   597→          }, { status: 200 });
   598→   598→        }
   599→   599→
   600→   600→        // Check for team limit reached
   601→   601→        if (lowerMsg.includes("limit") || lowerMsg.includes("maximum") || lowerMsg.includes("already")) {
   602→   602→          return NextResponse.json({
   603→   603→            status: "fail",
   604→   604→            error: `Team limit reached on ${app}. You already have the maximum number of teams for this match.`,
   605→   605→            code: "TEAM_LIMIT_REACHED",
   606→   606→            backendError: lastError,
   607→   607→          }, { status: 200 });
   608→   608→        }
   609→   609→
   610→   610→        // For 404, skip to next endpoint
   611→   611→        if (res.status === 404) {
   612→   612→          console.log(`[Transfer][SKIP] Endpoint returned 404. Trying next...`);
   613→   613→          continue;
   614→   614→        }
   615→   615→
   616→   616→        // For other errors, try next endpoint
   617→   617→        continue;
   618→   618→
   619→   619→      } catch (fetchErr) {
   620→   620→        const errMsg = fetchErr instanceof Error ? fetchErr.message : String(fetchErr);
   621→   621→        console.error(`[Transfer][NETWORK] Fetch error: ${errMsg}`);
   622→   622→        lastError = errMsg.includes("timeout") ? "Transfer request timed out (30s)" :
   623→   623→                    errMsg.includes("abort") ? "Request was aborted" :
   624→   624→                    `Network error: ${errMsg}`;
   625→   625→        lastHttpStatus = 0;
   626→   626→        continue;
   627→   627→      }
   628→   628→    }
   629→   629→
   630→   630→    // If edit failed, try add as fallback
   631→   631→    if (operation === "edit" && lastError && !isConfirmedTokenExpiry(lastError, lastData || {})) {
   632→   632→      console.log(`[Transfer][FALLBACK] Edit failed on all endpoints. Trying add as fallback...`);
   633→   633→
   634→   634→      const fallbackPayload = { ...payload };
   635→   635→      delete fallbackPayload.id;
   636→   636→      fallbackPayload.type = "new";
   637→   637→
   638→   638→      const addEndpoints = config.addEndpoints;
   639→   639→
   640→   640→      for (const fallbackUrl of addEndpoints) {
   641→   641→        console.log(`[Transfer][FALLBACK] Trying add endpoint: ${fallbackUrl}`);
   642→   642→
   643→   643→        try {
   644→   644→          const fallbackStart = Date.now();
   645→   645→          const fallbackRes = await fetch(fallbackUrl, {
   646→   646→            method: "POST",
   647→   647→            headers: { "Content-Type": "application/json" },
   648→   648→            body: JSON.stringify(fallbackPayload),
   649→   649→            signal: AbortSignal.timeout(30000),
   650→   650→          });
   651→   651→          const fallbackDuration = Date.now() - fallbackStart;
   652→   652→
   653→   653→          const fallbackText = await fallbackRes.text();
   654→   654→
   655→   655→          console.log(`[Transfer][FALLBACK] HTTP ${fallbackRes.status}, duration: ${fallbackDuration}ms`);
   656→   656→          console.log(`[Transfer][FALLBACK] Response: ${fallbackText.substring(0, 500)}`);
   657→   657→
   658→   658→          let fallbackData: Record<string, unknown>;
   659→   659→          try {
   660→   660→            fallbackData = JSON.parse(fallbackText);
   661→   661→          } catch {
   662→   662→            console.error(`[Transfer][FALLBACK] Non-JSON response`);
   663→   663→            fallbackData = {};
   664→   664→          }
   665→   665→
   666→   666→          // Use improved success check
   667→   667→          const fallbackCheck = isTransferSuccess(fallbackRes.status, fallbackData);
   668→   668→          if (fallbackCheck.success) {
   669→   669→            let responseData = fallbackData.data;
   670→   670→            if (typeof fallbackData.data === "string" && fallbackData.data) {
   671→   671→              const decrypted = decryptAES(fallbackData.data);
   672→   672→              if (decrypted) {
   673→   673→                try { responseData = JSON.parse(decrypted); } catch { responseData = { rawDecrypted: decrypted }; }
   674→   674→              }
   675→   675→            }
   676→   676→
   677→   677→            console.log(`[Transfer][FALLBACK][SUCCESS] Team added via fallback (confidence: ${fallbackCheck.confidence})`);
   678→   678→
   679→   679→            return NextResponse.json({
   680→   680→              status: "success",
   681→   681→              method: "fallback_add",
   682→   682→              endpoint: fallbackUrl,
   683→   683→              platform: app,
   684→   684→              operation: "add",
   685→   685→              data: responseData || null,
   686→   686→              httpStatus: fallbackRes.status,
   687→   687→              confidence: fallbackCheck.confidence,
   688→   688→              message: fallbackCheck.confidence === 'tentative'
   689→   689→                ? `Team added to ${app} (verified by proxy, fallback)`
   690→   690→                : `Team added to ${app} successfully (fallback from edit)`,
   691→   691→            }, { status: 200 });
   692→   692→          }
   693→   693→
   694→   694→          const fallbackError = (fallbackData?.message as string) || (fallbackData?.error as string) || "Unknown fallback error";
   695→   695→          console.error(`[Transfer][FALLBACK][FAILED] ${fallbackError}`);
   696→   696→
   697→   697→          // Check for token expiry in fallback too
   698→   698→          if (isConfirmedTokenExpiry(fallbackError, fallbackData)) {
   699→   699→            return NextResponse.json({
   700→   700→              status: "fail",
   701→   701→              error: `Session expired on ${app}. Please re-link your account via OTP verification.`,
   702→   702→              code: "TOKEN_EXPIRED",
   703→   703→              needsReauth: true,
   704→   704→              backendError: fallbackError,
   705→   705→            }, { status: 200 });
   706→   706→          }
   707→   707→
   708→   708→          if (fallbackError && fallbackError !== "Something Went Wrong!") {
   709→   709→            lastError = fallbackError;
   710→   710→          }
   711→   711→          lastHttpStatus = fallbackRes.status;
   712→   712→        } catch (fallbackErr) {
   713→   713→          console.error(`[Transfer][FALLBACK][ERROR] ${fallbackErr instanceof Error ? fallbackErr.message : "unknown"}`);
   714→   714→        }
   715→   715→      }
   716→   716→    }
   717→   717→
   718→   718→    // Return detailed error
   719→   719→    console.error(`\n${"!".repeat(60)}`);
   720→   720→    console.error(`[Transfer][FINAL] ALL ENDPOINTS FAILED for ${app} ${operation}`);

   722→   722→    console.error(`[Transfer][FINAL] HTTP status: ${lastHttpStatus}`);
   723→   723→    console.error(`[Transfer][FINAL] Endpoints tried: ${endpointChain.join(", ")}`);

   726→   726→
   727→   727→    let displayError = lastError || "Unknown transfer error";
   728→   728→

   733→   733→    } else if (displayError === "Proxy returned 400") {
   734→   734→      displayError = `Your ${app} session has expired (the platform rejected the request). Please re-link your account via OTP verification.`;
   735→   735→    } else if (displayError === "Error while transfering the team!") {

   737→   737→    } else if (displayError.includes("non-JSON response") && lastHttpStatus === 0) {

   739→   739→    }
   740→   740→
   741→   741→    return NextResponse.json({
   742→   742→      status: "fail",
   743→   743→      error: displayError,
   744→   744→      code: "TRANSFER_ERROR",
   745→   745→      needsReauth: false,
   746→   746→      backendError: lastError,
   747→   747→      httpStatus: lastHttpStatus,
   748→   748→      platform: app,
   749→   749→      operation: operation,
   750→   750→      endpointsTried: endpointChain,
   751→   751→      details: {
   752→   752→        matchId: String(matchId),
   753→   753→        captain: finalCaptain,
   754→   754→        vice_captain: finalViceCaptain,
   755→   755→        playerCount: numericPlayerIds.length,
   756→   756→        teamId: id || null,
   757→   757→        tokenLength: preparedToken.length,
   758→   758→        tokenIsJWT: preparedToken.startsWith("eyJ"),
   759→   759→        tokenIsJSON: preparedToken.startsWith("{"),
   760→   760→      },
   761→   761→    }, { status: 200 });
   762→   762→
   763→   763→  } catch (error) {
   764→   764→    console.error("[Transfer][ROUTE_ERROR]", error);
   765→   765→    return NextResponse.json({
   766→   766→      status: "fail",
   767→   767→      error: `Server error: ${error instanceof Error ? error.message : "Unknown error"}`,
   768→   768→      code: "SERVER_ERROR",
   769→   769→      needsReauth: false,
   770→   770→    }, { status: 200 });
   771→   771→  }
