   205→   205→   205→function randomKeySegment(): string {
   206→   206→   206→  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
   207→   207→   207→  let out = "";
   208→   208→   208→  for (let i = 0; i < 4; i++) {
   209→   209→   209→    out += chars[Math.floor(Math.random() * chars.length)];
   210→   210→   210→  }
   211→   211→   211→  return out;
   212→   212→   212→}

   240→   240→   240→function lsGet<T>(key: string, fallback: T): T {
   241→   241→   241→  if (typeof window === "undefined") return fallback;
   242→   242→   242→  try {
   243→   243→   243→    const raw = localStorage.getItem(key);
   244→   244→   244→    if (!raw) return fallback;
   245→   245→   245→    return JSON.parse(raw) as T;
   246→   246→   246→  } catch {
   247→   247→   247→    return fallback;
   248→   248→   248→  }
   249→   249→   249→}

   257→   257→   257→  }
