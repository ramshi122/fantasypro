   128→function scorePlayer(p: Player) {
   129→  const statusBoost = p.status === "playing" ? 35 : p.status === "sub" ? 22 : 8;
   130→  const roleBoost = roleRank(p.role) * 6;
   131→  const pickBoost = p.fixed ? 18 : p.cap ? 16 : p.vc ? 12 : 0;
   132→  const selection = Math.min(30, numVal(p.selPct, 0) * 0.3);
   133→  return statusBoost + roleBoost + pickBoost + selection;
   134→}

   136→function buildFallbackTeams(players: Player[], isFive: boolean, team1: string, team2: string): GenTeam[] {
   137→  const size = isFive ? 5 : 11;
   138→  const numTeams = isFive ? 4 : 8;
   139→  const pool = [...players]
   140→    .filter((p) => p.status !== "bench" || players.filter((x) => x.status !== "bench").length < size)
   141→    .sort((a, b) => scorePlayer(b) - scorePlayer(a));
   142→  const locked = pool.filter((p) => p.fixed || p.cap).slice(0, Math.max(0, size - 2));
   143→  const captainPool = pool.filter((p) => p.cap || p.fixed).concat(pool).filter(Boolean);
   144→  const vcPool = pool.filter((p) => p.vc || p.fixed).concat(pool).filter(Boolean);
   145→  const styles = ["SAFE", "GL", "PITCH", "TOSS", "IMPACT", "DEATH", "TOP", "CONTRA"];
   146→
   147→  return Array.from({ length: numTeams }, (_, i) => {
   148→    const chosen = new Map<string, Player>();
   149→    locked.forEach((p) => chosen.set(p.name, p));
   150→    const rotated = [...pool.slice(i), ...pool.slice(0, i)];
   151→    for (const role of ["WK", "AR", "BAT", "BOWL"] as const) {
   152→      if (chosen.size >= size) break;
   153→      const pick = rotated.find((p) => p.role === role && !chosen.has(p.name));
   154→      if (pick) chosen.set(pick.name, pick);
   155→    }
   156→    for (const p of rotated) {
   157→      if (chosen.size >= size) break;
   158→      if (!chosen.has(p.name)) chosen.set(p.name, p);
   159→    }
   160→    const selected = Array.from(chosen.values()).slice(0, size);
   161→    const c = captainPool[i % Math.max(1, captainPool.length)] || selected[0];
   162→    const vc = vcPool[(i + 1) % Math.max(1, vcPool.length)] || selected[1] || selected[0];
   163→    const finalPlayers = selected.map((p, j) => ({
   164→      name: p.name, role: p.role,
   165→      isC: p.name === c?.name,
   166→      isVC: p.name === vc?.name && p.name !== c?.name,
   167→      impact: j >= Math.max(0, size - 2),
   168→    }));
   169→    return {
   170→      title: `${styles[i % styles.length]} ${isFive ? "5s" : "11s"} · ${team1} vs ${team2}`,
   171→      style: styles[i % styles.length],
   172→      players: finalPlayers,
   173→      reason: "Fallback hack team from verified scraped pool. Prioritising playing/sub players, locked picks, role balance and rotating C/VC combos.",
   174→      confidence: 72,
   175→      isFive,
   176→    };
   177→  }).filter((team) => team.players.length === size);
   178→}

   180→function buildFallbackDt(players: Player[], team1: string, team2: string) {
   181→  const ranked = [...players].sort((a, b) => scorePlayer(b) - scorePlayer(a)).map((p, i) => {
   182→    const dtPct = Math.max(8, Math.min(92, Math.round(scorePlayer(p) - i * 1.4)));
   183→    return {
   184→      name: p.name,
   185→      team: teamLabel(p.team, team1, team2),
   186→      role: p.role,
   187→      selPct: Math.round(numVal(p.selPct, 0)),
   188→      dtPct,
   189→      avg2026: Math.max(12, Math.round(dtPct * 0.72 + roleRank(p.role) * 3)),
   190→      reason: `${p.status.toUpperCase()} pool · role ${p.role} · calculated from scraped lineup.`,
   191→    } satisfies DtPlayer;
   192→  });
   193→  const split = Math.max(1, Math.ceil(ranked.length * 0.6));
   194→  return { likely: ranked.slice(0, split), avoid: ranked.slice(split), total: ranked.length };
   195→}

   197→function extractJSON(raw: string): any {
   198→  let cleaned = raw.replace(/```json\s*/gi, "").replace(/```/g, "").trim();
   199→  if (!cleaned.startsWith("{") && !cleaned.startsWith("[")) {
   200→    const o = cleaned.indexOf("{");
   201→    const a = cleaned.indexOf("[");
   202→    const isArr = a !== -1 && (o === -1 || a < o);
   203→    const start = isArr ? a : o;
   204→    const end = isArr ? cleaned.lastIndexOf("]") : cleaned.lastIndexOf("}");
   205→    if (start !== -1 && end > start) cleaned = cleaned.slice(start, end + 1);
   206→    else throw new Error("No JSON found");
   207→  }
   208→  return JSON.parse(cleaned);
   209→}

   263→function fileToBase64(file: File): Promise<string> {
   264→  return new Promise((resolve, reject) => {
   265→    const reader = new FileReader();
   266→    reader.onload = () => {
   267→      const img = new Image();
   268→      img.onload = () => {
   269→        let { width, height } = img;
   270→        const scale = Math.min(1, 1280 / Math.max(width, height));
   271→        width = Math.round(width * scale);
   272→        height = Math.round(height * scale);
   273→        const canvas = document.createElement("canvas");
   274→        canvas.width = width;
   275→        canvas.height = height;
   276→        const ctx = canvas.getContext("2d");
   277→        if (!ctx) return reject(new Error("canvas failed"));
   278→        ctx.drawImage(img, 0, 0, width, height);
   279→        resolve(canvas.toDataURL("image/jpeg", 0.78));
   280→      };
   281→      img.onerror = reject;
   282→      img.src = reader.result as string;
   283→    };
   284→    reader.onerror = reject;
   285→    reader.readAsDataURL(file);
   286→  });
   287→}
