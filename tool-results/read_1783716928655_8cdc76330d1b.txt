   131→/** Deterministic hash-based PRNG (mulberry32). Same seed → same sequence. */
   132→function mulberry32(seed: number): () => number {
   133→  let a = seed >>> 0;
   134→  return function () {
   135→    a |= 0;
   136→    a = (a + 0x6d2b79f5) | 0;
   137→    let t = a;
   138→    t = Math.imul(t ^ (t >>> 15), t | 1);
   139→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
   140→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   141→  };
   142→}

   144→function clamp(n: number, min = 0, max = 100): number {
   145→  return Math.max(min, Math.min(max, n));
   146→}
   147→
   148→function avg(nums: number[]): number {
   149→  if (nums.length === 0) return 0;
   150→  return nums.reduce((s, n) => s + n, 0) / nums.length;
   151→}

   153→/* ============== Player EFV (Expected Fantasy Value) ============== */
   154→
   155→/**
   156→ * Calculate EFV for a single player.
   157→ *
   158→ * EFV = base_value × form_factor × role_weight × venue_factor
   159→ *
   160→ * - base_value: derived from sel% (crowd wisdom)
   161→ * - form_factor: derived from credits (higher credits = better recent form)
   162→ * - role_weight: AR gets bonus (dual scoring), BAT/BOWL normal, WK slight bonus
   163→ * - venue_factor: not available without venue data, defaults to 1.0
   164→ */
   165→export function calculatePlayerEFV(p: MatchPlayer): number {
   166→  const sel = selPct(p);
   167→  const cred = credits(p);
   168→  const role = roleOf(p);
   169→
   170→  // Base value from selection % (normalized 0-1)
   171→  const baseValue = Math.log(1 + sel) / Math.log(101); // log scale: 0% → 0, 100% → 1
   172→
   173→  // Form factor from credits (8.0-10.0 = good form, 8.5-9.5 = peak)
   174→  const formFactor = clamp((cred - 7) / 3, 0.3, 1.5);
   175→
   176→  // Role weight
   177→  const roleWeights: Record<string, number> = {
   178→    AR: 1.25,   // all-rounders score from both batting + bowling
   179→    WK: 1.10,   // WK gets batting + keeping bonus
   180→    BAT: 1.00,
   181→    BOWL: 1.00,
   182→  };
   183→  const roleWeight = roleWeights[role] ?? 1.0;
   184→
   185→  // Differential bonus: low-sel% players get a small upside bonus (GL value)
   186→  const diffBonus = sel < 15 ? 1.10 : sel < 30 ? 1.05 : 1.00;
   187→
   188→  return baseValue * formFactor * roleWeight * diffBonus * 100;
   189→}

   193→/**
   194→ * Safe math helpers — prevent NaN/Infinity crashes on empty arrays.
   195→ */
   196→function safeMax(nums: number[]): number {
   197→  if (nums.length === 0) return 0;
   198→  return Math.max(...nums);
   199→}
   200→function safeMin(nums: number[]): number {
   201→  if (nums.length === 0) return 0;
   202→  return Math.min(...nums);
   203→}

   205→export function runEliteAnalysis(
   206→  players: MatchPlayer[],
   207→  team1Name: string,
   208→  team2Name: string,
   209→  lineupAnnounced: boolean = false
   210→): EliteAnalysisResult {
   211→  const checks: EliteAnalysisCheck[] = [];
   212→  const pool = (players || []).filter((p) => p && p.name && credits(p) > 0);
   213→  const playing = pool.filter((p) => p.playing === 1 || p.playing === undefined || p.playing === null);
   214→
   215→  // If pool is too small, return early with safe defaults
   216→  if (pool.length === 0) {
   217→    return {
   218→      checks: [{
   219→        id: 150,
   220→        name: "Final Team Optimization Score",
   221→        category: "confidence",
   222→        group: "optimization",
   223→        score: 0,
   224→        status: "critical",
   225→        insight: "No valid players in pool — cannot analyze",
   226→      }],
   227→      overallScore: 0,
   228→      confidence: "low",
   229→      recommendedStrategy: "safe",
   230→      poolStats: {
   231→        totalPlayers: 0, playingXI: 0, wkCount: 0, batCount: 0, arCount: 0, bowlCount: 0,
   232→        avgCredits: 0, avgSelPct: 0,
   233→      },
   234→      topCaptainPicks: [],
   235→      topVCPicks: [],
   236→      differentialPicks: [],
   237→    };
   238→  }
   239→
   240→  // Group by role
   241→  const byRole = {
   242→    WK: pool.filter((p) => roleOf(p) === "WK"),
   243→    BAT: pool.filter((p) => roleOf(p) === "BAT"),
   244→    AR: pool.filter((p) => roleOf(p) === "AR"),
   245→    BOWL: pool.filter((p) => roleOf(p) === "BOWL"),
   246→  };
   247→
   248→
   249→  const allSelPct = pool.map(selPct);
   250→  const allCredits = pool.map(credits);
   251→
   252→  // === CHECK 1: Player Pool Depth ===
   253→  const minPerRole = Math.min(byRole.WK.length, byRole.BAT.length, byRole.AR.length, byRole.BOWL.length);
   254→  const poolDepthScore = clamp(minPerRole * 25, 0, 100);
   255→  checks.push({
   256→    id: 1,
   257→    name: "Player Pool Depth",
   258→    category: "pool",
   259→    group: "match",
   260→    score: poolDepthScore,
   261→    status: minPerRole >= 4 ? "excellent" : minPerRole >= 3 ? "good" : minPerRole >= 2 ? "warning" : "critical",
   262→    insight: `WK:${byRole.WK.length} BAT:${byRole.BAT.length} AR:${byRole.AR.length} BOWL:${byRole.BOWL.length} — min ${minPerRole} per role`,
   263→    recommendation: minPerRole < 3 ? "Pool too shallow — generation may fail" : undefined,
   264→  });
   265→
   266→  // === CHECK 2: Playing XI Confirmation ===
   267→  const playingCount = playing.length;
   268→  const lineupScore = lineupAnnounced ? 100 : playingCount >= 22 ? 80 : playingCount >= 16 ? 60 : 30;
   269→  checks.push({
   270→    id: 2,
   271→    name: "Playing XI Confirmation",
   272→    category: "lineup",
   273→    group: "match",
   274→    score: lineupScore,
   275→    status: lineupAnnounced ? "excellent" : playingCount >= 22 ? "good" : "warning",
   276→    insight: lineupAnnounced ? "Lineup announced — only playing XI used" : `${playingCount} players in pool — lineup not confirmed`,
   277→    recommendation: !lineupAnnounced ? "Wait for lineup for best accuracy" : undefined,
   278→  });
   279→
   280→  // === CHECK 3: Captain Certainty ===
   281→  const sortedBySel = [...pool].sort((a, b) => selPct(b) - selPct(a));
   282→  const topSel = selPct(sortedBySel[0] || { selected_by: 0 } as MatchPlayer);
   283→  const secondSel = selPct(sortedBySel[1] || { selected_by: 0 } as MatchPlayer);
   284→  const capGap = topSel - secondSel;
   285→  const capScore = clamp(capGap * 2 + 30, 0, 100);
   286→  checks.push({
   287→    id: 3,
   288→    name: "Captain Certainty",
   289→    category: "captain",
   290→    group: "team",
   291→    score: capScore,
   292→    status: capGap > 20 ? "excellent" : capGap > 10 ? "good" : capGap > 5 ? "warning" : "critical",
   293→    insight: `Top pick ${topSel.toFixed(1)}% vs #2 ${secondSel.toFixed(1)}% — gap ${capGap.toFixed(1)}%`,
   294→    recommendation: capGap < 5 ? "No clear captain — rotate C across teams" : undefined,
   295→  });
   296→
   297→  // === CHECK 4: Vice-Captain Certainty ===
   298→  const thirdSel = selPct(sortedBySel[2] || { selected_by: 0 } as MatchPlayer);
   299→  const vcGap = secondSel - thirdSel;
   300→  const vcScore = clamp(vcGap * 3 + 40, 0, 100);
   301→  checks.push({
   302→    id: 4,
   303→    name: "Vice-Captain Certainty",
   304→    category: "captain",
   305→    group: "team",
   306→    score: vcScore,
   307→    status: vcGap > 10 ? "excellent" : vcGap > 5 ? "good" : "warning",
   308→    insight: `VC candidates: ${secondSel.toFixed(1)}% vs #3 ${thirdSel.toFixed(1)}% — gap ${vcGap.toFixed(1)}%`,
   309→  });
   310→
   311→  // === CHECK 5: Credit Distribution ===
   312→  const avgCred = avg(allCredits);
   313→  const credSpread = safeMax(allCredits) - safeMin(allCredits);
   314→  const credScore = clamp(credSpread >= 4 && credSpread <= 6 ? 100 : credSpread >= 3 ? 80 : 50, 0, 100);
   315→  checks.push({
   316→    id: 5,
   317→    name: "Credit Distribution",
   318→    category: "balance",
   319→    group: "team",
   320→    score: credScore,
   321→    status: credScore >= 80 ? "excellent" : credScore >= 60 ? "good" : "warning",
   322→    insight: `Avg ${avgCred.toFixed(1)} credits · spread ${credSpread.toFixed(1)} · range ${safeMin(allCredits).toFixed(1)}-${safeMax(allCredits).toFixed(1)}`,
   323→  });
   324→
   325→  // === CHECK 6: Selection % Concentration ===
   326→  const highSelCount = pool.filter((p) => selPct(p) >= 50).length;
   327→  const concScore = clamp(highSelCount >= 6 && highSelCount <= 10 ? 100 : highSelCount >= 4 ? 80 : 50, 0, 100);
   328→  checks.push({
   329→    id: 6,
   330→    name: "Selection % Concentration",
   331→    category: "balance",
   332→    group: "team",
   333→    score: concScore,
   334→    status: concScore >= 80 ? "excellent" : "good",
   335→    insight: `${highSelCount} players with ≥50% selection — ${highSelCount > 10 ? "too concentrated" : "balanced"}`,
   336→  });
   337→
   338→  // === CHECK 7: Differential Opportunity ===
   339→  const diffCount = pool.filter((p) => selPct(p) < 20 && selPct(p) > 0).length;
   340→  const diffScore = clamp(diffCount >= 5 ? 100 : diffCount >= 3 ? 80 : diffCount >= 1 ? 60 : 30, 0, 100);
   341→  checks.push({
   342→    id: 7,
   343→    name: "Differential Opportunity",
   344→    category: "risk",
   345→    group: "team",
   346→    score: diffScore,
   347→    status: diffCount >= 5 ? "excellent" : diffCount >= 3 ? "good" : "warning",
   348→    insight: `${diffCount} differential picks (<20% sel) available for GL upside`,
   349→    recommendation: diffCount < 3 ? "Limited differentials — GL teams may look similar" : undefined,
   350→  });
   351→
   352→  // === CHECK 8: Team Balance (7-4 vs 6-5) ===
   353→  const t1Count = pool.filter((p) => (p.team_name || "") === team1Name).length;
   354→  const t2Count = pool.filter((p) => (p.team_name || "") === team2Name).length;
   355→  const balanceGap = Math.abs(t1Count - t2Count);
   356→  const balanceScore = clamp(100 - balanceGap * 8, 0, 100);
   357→  checks.push({
   358→    id: 8,
   359→    name: "Team Balance",
   360→    category: "balance",
   361→    group: "team",
   362→    score: balanceScore,
   363→    status: balanceGap <= 2 ? "excellent" : balanceGap <= 4 ? "good" : "warning",
   364→    insight: `${team1Name}: ${t1Count} players vs ${team2Name}: ${t2Count} players — gap ${balanceGap}`,
   365→  });
   366→
   367→  // === CHECK 9: Role Balance Optimal ===
   368→  const roleBalanceOk = byRole.WK.length >= 2 && byRole.BAT.length >= 4 && byRole.AR.length >= 2 && byRole.BOWL.length >= 4;
   369→  const roleScore = clamp(roleBalanceOk ? 100 : 60, 0, 100);
   370→  checks.push({
   371→    id: 9,
   372→    name: "Role Balance Optimal",
   373→    category: "balance",
   374→    group: "team",
   375→    score: roleScore,
   376→    status: roleBalanceOk ? "excellent" : "warning",
   377→    insight: `Need ≥2 WK, ≥4 BAT, ≥2 AR, ≥4 BOWL for diverse combinations`,
   378→  });
   379→
   380→  // === CHECK 10: Top-Order BAT Strength ===
   381→  const topBats = byRole.BAT.sort((a, b) => selPct(b) - selPct(a)).slice(0, 3);
   382→  const batStrength = avg(topBats.map(selPct));
   383→  const batScore = clamp(batStrength * 1.5, 0, 100);
   384→  checks.push({
   385→    id: 10,
   386→    name: "Top-Order BAT Strength",
   387→    category: "form",
   388→    group: "team",
   389→    score: batScore,
   390→    status: batStrength >= 50 ? "excellent" : batStrength >= 30 ? "good" : "warning",
   391→    insight: `Top 3 batters avg ${batStrength.toFixed(1)}% selection`,
   392→  });
   393→
   394→  // === CHECK 11: Death Overs BOWL Strength ===
   395→  const topBowlers = byRole.BOWL.sort((a, b) => selPct(b) - selPct(a)).slice(0, 3);
   396→  const bowlStrength = avg(topBowlers.map(selPct));
   397→  const bowlScore = clamp(bowlStrength * 1.5, 0, 100);
   398→  checks.push({
   399→    id: 11,
   400→    name: "Death Overs BOWL Strength",
   401→    category: "form",
   402→    group: "team",
   403→    score: bowlScore,
   404→    status: bowlStrength >= 50 ? "excellent" : bowlStrength >= 30 ? "good" : "warning",
   405→    insight: `Top 3 bowlers avg ${bowlStrength.toFixed(1)}% selection`,
   406→  });
   407→
   408→  // === CHECK 12: All-Rounder Coverage ===
   409→  const topARs = byRole.AR.sort((a, b) => selPct(b) - selPct(a));
   410→  const arScore = clamp(topARs.length >= 3 ? 100 : topARs.length >= 2 ? 80 : 50, 0, 100);
   411→  checks.push({
   412→    id: 12,
   413→    name: "All-Rounder Coverage",
   414→    category: "balance",
   415→    group: "team",
   416→    score: arScore,
   417→    status: topARs.length >= 3 ? "excellent" : "good",
   418→    insight: `${topARs.length} ARs available — top AR sel%: ${topARs[0] ? selPct(topARs[0]).toFixed(1) : "N/A"}%`,
   419→    recommendation: topARs.length < 2 ? "Add more ARs for balanced teams" : undefined,
   420→  });
   421→
   422→  // === CHECK 13: Wicket-Keeper Form ===
   423→  const topWK = byRole.WK.sort((a, b) => selPct(b) - selPct(a))[0];
   424→  const wkScore = clamp(topWK ? selPct(topWK) * 1.5 : 30, 0, 100);
   425→  checks.push({
   426→    id: 13,
   427→    name: "Wicket-Keeper Form",
   428→    category: "form",
   429→    group: "team",
   430→    score: wkScore,
   431→    status: wkScore >= 60 ? "excellent" : wkScore >= 40 ? "good" : "warning",
   432→    insight: topWK ? `Top WK: ${playerName(topWK)} (${selPct(topWK).toFixed(1)}% sel)` : "No WK available",
   433→  });
   434→
   435→  // === CHECK 14: Player Form Trend ===
   436→  const highCredCount = pool.filter((p) => credits(p) >= 9).length;
   437→  const formScore = clamp(highCredCount >= 5 ? 100 : highCredCount >= 3 ? 80 : 50, 0, 100);
   438→  checks.push({
   439→    id: 14,
   440→    name: "Player Form Trend",
   441→    category: "form",
   442→    group: "team",
   443→    score: formScore,
   444→    status: formScore >= 80 ? "excellent" : "good",
   445→    insight: `${highCredCount} players in peak form (≥9.0 credits)`,
   446→  });
   447→
   448→  // === CHECK 15: Venue Suitability ===
   449→  const batVsBowlSel = avg(byRole.BAT.map(selPct)) - avg(byRole.BOWL.map(selPct));
   450→  const venueScore = clamp(70 + Math.abs(batVsBowlSel) * 0.5, 0, 100);
   451→  checks.push({
   452→    id: 15,
   453→    name: "Venue Suitability",
   454→    category: "venue",
   455→    group: "match",
   456→    score: venueScore,
   457→    status: "good",
   458→    insight: batVsBowlSel > 5 ? "Venue appears batting-friendly" : batVsBowlSel < -5 ? "Venue appears bowling-friendly" : "Balanced venue",
   459→  });
   460→
   461→  // === CHECK 16: Toss Impact ===
   462→  const tossScore = clamp(lineupAnnounced ? 100 : 50, 0, 100);
   463→  checks.push({
   464→    id: 16,
   465→    name: "Toss Impact",
   466→    category: "venue",
   467→    group: "match",
   468→    score: tossScore,
   469→    status: lineupAnnounced ? "excellent" : "warning",
   470→    insight: lineupAnnounced ? "Toss done — teams locked" : "Toss pending — team composition may change",
   471→    recommendation: !lineupAnnounced ? "Generate after toss for best results" : undefined,
   472→  });
   473→
   474→  // === CHECK 17: Injury/Bench Risk ===
   475→  const nonPlaying = pool.length - playing.length;
   476→  const riskScore = clamp(nonPlaying === 0 ? 100 : 100 - nonPlaying * 5, 0, 100);
   477→  checks.push({
   478→    id: 17,
   479→    name: "Injury/Bench Risk",
   480→    category: "risk",
   481→    group: "match",
   482→    score: riskScore,
   483→    status: nonPlaying === 0 ? "excellent" : nonPlaying <= 2 ? "good" : "warning",
   484→    insight: `${nonPlaying} players at risk of being benched`,
   485→  });
   486→
   487→  // === CHECK 18: Captain Multiplier Effect ===
   488→  const topEFV = [...pool].sort((a, b) => calculatePlayerEFV(b) - calculatePlayerEFV(a))[0];
   489→  const capMultScore = clamp(topEFV ? calculatePlayerEFV(topEFV) : 30, 0, 100);
   490→  checks.push({
   491→    id: 18,
   492→    name: "Captain Multiplier Effect",
   493→    category: "captain",
