     4→ * Hidden admin mode with license management, device binding, user control,

     7→ * Storage model (per FIREBASE-CLEANUP):

    18→ * When admin CHECKS keys → read from localStorage (seed + generated keys).
    19→ *
    20→ * When a USER verifies a key → read ONE doc from Firebase by key string
    21→ * (NOT a collection download) — handled in `verifyLicense` via
    22→ * `fetchKeyFromFirestore`.
    23→ *
    24→ * All real-time `onSnapshot` listeners have been REMOVED.
    25→ * One-shot reads/writes only.

   205→function randomKeySegment(): string {
   206→  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
   207→  let out = "";
   208→  for (let i = 0; i < 4; i++) {
   209→    out += chars[Math.floor(Math.random() * chars.length)];
   210→  }
   211→  return out;
   212→}

   218→/**
   219→ * Get expiry timestamp for a key type.
   220→ *
   221→ * Uses server time if available (from localStorage cache set by
   222→ * admin-realtime-sync onSnapshot listener), otherwise falls back
   223→ * to device time.
   224→ *
   225→ * This ensures keys generated on different devices have consistent
   226→ * expiry times (within 30 seconds of each other).
   227→ */
   228→export function getExpiryForType(type: KeyType): number | null {
   229→  // Try to use cached server time
   230→  let now = Date.now();
   231→  if (typeof window !== "undefined") {
   232→    try {
   233→      const lsTime = localStorage.getItem("apna-server-time");
   234→      const lsFetched = localStorage.getItem("apna-server-time-fetched");
   235→      if (lsTime && lsFetched) {
   236→        const serverTime = Number(lsTime);
   237→        const fetchedAt = Number(lsFetched);
   238→        // If cached within last 60 seconds, use server time + elapsed
   239→        if (Date.now() - fetchedAt < 60_000) {
   240→          now = serverTime + (Date.now() - fetchedAt);
   241→        }
   242→      }
   243→    } catch {
   244→      // ignore
   245→    }
   246→  }
   247→
   248→  const day = 86400000;
   249→  const hour = 3600000;
   250→  switch (type) {
   251→    case "trial": return now + 1 * day;          // 1 day free
   252→    case "match_basic": return now + 6 * hour;   // 1 match = 6 hours
   253→    case "match_full": return now + 1 * day;     // 24 hour active
   254→    case "weekly": return now + 7 * day;         // 7 days
   255→    case "monthly": return now + 30 * day;       // 30 days (Pro ₹299)
   256→    case "half_yearly": return now + 180 * day;  // 180 days (Semi-Annual ₹499)
   257→    case "elite": return now + 90 * day;         // 90 days (Elite ₹999)
   258→    case "lifetime": return null;                // forever
   259→  }
   260→}

   268→function lsGet<T>(key: string, fallback: T): T {
   269→  if (typeof window === "undefined") return fallback;
   270→  try {
   271→    const raw = localStorage.getItem(key);
   272→    if (!raw) return fallback;
   273→    return JSON.parse(raw) as T;
   274→  } catch {
   275→    return fallback;
   276→  }
   277→}
   278→
   279→function lsSet<T>(key: string, value: T): void {
   280→  if (typeof window === "undefined") return;
   281→  try {
   282→    localStorage.setItem(key, JSON.stringify(value));
   283→  } catch {
   284→    /* ignore quota errors */
   285→  }
   286→}

   336→/**
   337→ * Bootstrap initial state from localStorage. If localStorage is empty for
   338→ * a given collection (first run on this device), seed it with the initial
   339→ * demo data so that EVERY phone sees the same set of seed keys + devices +
   340→ * users + logs.
   341→ */
   342→function bootstrapInitialState() {
   343→  const initial = getInitialData();
   344→
   345→  const persistedKeys = lsGet<LicenseKey[] | null>(LS_KEYS, null);
   346→  const keys = persistedKeys && persistedKeys.length > 0 ? persistedKeys : initial.keys;
   347→  if (!persistedKeys || persistedKeys.length === 0) lsSet(LS_KEYS, keys);
   348→
   349→  const persistedDevices = lsGet<Device[] | null>(LS_DEVICES, null);
   350→  const devices = persistedDevices && persistedDevices.length > 0 ? persistedDevices : initial.devices;
   351→  if (!persistedDevices || persistedDevices.length === 0) lsSet(LS_DEVICES, devices);
   352→
   353→  const persistedUsers = lsGet<User[] | null>(LS_USERS, null);
   354→  const users = persistedUsers && persistedUsers.length > 0 ? persistedUsers : initial.users;
   355→  if (!persistedUsers || persistedUsers.length === 0) lsSet(LS_USERS, users);
   356→
   357→  const persistedLogs = lsGet<ActivityLog[] | null>(LS_LOGS, null);
   358→  const logs = persistedLogs && persistedLogs.length > 0 ? persistedLogs : initial.logs;
   359→  if (!persistedLogs || persistedLogs.length === 0) lsSet(LS_LOGS, logs);
   360→
   361→  const persistedSettings = lsGet<AdminSettings | null>(LS_SETTINGS, null);
   362→  const settings = persistedSettings ?? DEFAULT_SETTINGS;
   363→  if (!persistedSettings) lsSet(LS_SETTINGS, settings);
   364→
   365→  const announcements = lsGet<Announcement[]>(LS_ANNOUNCEMENTS, []);
   366→
   367→  return { keys, devices, users, logs, settings, announcements };
   368→}

  1120→  if (!id) {

  1127→export function getLicenseState(): LicenseState {

  1130→    const data = JSON.parse(localStorage.getItem(LICENSE_STORAGE_KEY) || "{}");
  1131→    return {
  1132→      verified: !!data.verified,
  1133→      key: data.key || null,
  1134→      deviceId: getDeviceId(),
  1135→      keyType: data.keyType || null,
  1136→    };
  1137→  } catch {
  1138→    return { verified: false, key: null, deviceId: getDeviceId(), keyType: null };
  1139→  }
