   167→function randomKeySegment(): string {
   168→  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
   169→  let out = "";
   170→  for (let i = 0; i < 4; i++) {
   171→    out += chars[Math.floor(Math.random() * chars.length)];
   172→  }
   173→  return out;
   174→}

   176→function generateLicenseKey(prefix: string = LICENSE_PREFIX_DEFAULT): string {
   177→  return `${prefix}-${randomKeySegment()}-${randomKeySegment()}-${randomKeySegment()}`;
   178→}

   180→export function getExpiryForType(type: KeyType): number | null {
   181→  const now = Date.now();
   182→  const day = 86400000;
   183→  const hour = 3600000;
   184→  switch (type) {
   185→    case "trial": return now + 1 * day;        // 1 day free
   186→    case "match_basic": return now + 6 * hour; // 1 match = 6 hours
   187→    case "match_full": return now + 1 * day;   // 24 hour active
   188→    case "weekly": return now + 7 * day;       // 7 days
   189→    case "monthly": return now + 30 * day;     // 30 days (Pro ₹299)
   190→    case "elite": return now + 90 * day;       // 90 days (Elite ₹999)
   191→    case "lifetime": return null;              // forever
   192→  }
   193→}

   195→function uid(): string {
   196→  return Math.random().toString(36).slice(2, 10) + Date.now().toString(36);

   434→            }

   438→              else if (k.type === "weekly") expiresAt = now + 7 * day;
   439→              else if (k.type === "monthly") expiresAt = now + 30 * day;
   440→              else if (k.type === "lifetime") expiresAt = null;
   441→              else expiresAt = now + 30 * day;
   442→              // status was already re-activated above if it was "expired"
   443→              fixedCount++;
   444→            }
   445→
   446→            return { ...k, status, expiresAt };
   447→          }),
   448→        }));
   449→

   471→        console.log("[License Verify] Device ID:", deviceId);

   474→        // Step 2: Validate format (CRICB-XXXX-XXXX-XXXX)

   478→          const state = recordFailedAttempt(cleanKey);
   479→          const remaining = getRemainingAttempts();
   480→          if (state.lockedUntil) {
   481→            return {
   482→              valid: false,
   483→              message: `🔒 5 wrong attempts reached. App locked for 1 hour. Try after 60 minutes.`,
   484→              locked: true,
   485→            } as any;
   486→          }
   487→          return { valid: false, message: `Invalid key format. ${remaining} attempt(s) remaining before 1-hour lock.` };
   488→        }
   489→
   490→        // Step 3: Look up key in LOCAL database
   491→        const allKeys = get().keys;
   492→        console.log("[License Verify] Total keys in local database:", allKeys.length);
   493→        let k = allKeys.find((x) => x.key.toUpperCase() === cleanKey);
   494→
   495→        // Step 3b: If not found locally, check FIREBASE (key may have been generated on another device)
   496→        if (!k) {
   497→          console.log("[License Verify] Key not found locally — checking Firebase...");
   498→          try {
   499→            // Dynamic import to avoid circular dependency
   500→            const { fetchKeyFromFirestore } = await import("./firebase-license");

   504→              // Add to local store for future lookups
   505→              set((s) => ({ keys: [firebaseKey, ...s.keys] }));
   506→              k = firebaseKey;
   507→            }
   508→          } catch (err) {
   509→            console.log("[License Verify] Firebase lookup failed:", err);
   510→          }
   511→        }
   512→
   513→        if (!k) {
   514→          console.log("[License Verify] FAIL: Key not found in local OR Firebase database");
   515→          const failState = recordFailedAttempt(cleanKey);
   516→          const remaining = getRemainingAttempts();
   517→          if (failState.lockedUntil) {
