   177→/* ---------- Login Modal ---------- */
   178→
   179→function AdminLoginModal({ open, onClose }: { open: boolean; onClose: () => void }) {
   180→  const admin = useAdmin();
   181→  const [password, setPassword] = useState("");
   182→  const [show, setShow] = useState(false);
   183→  const [loading, setLoading] = useState(false);
   184→
   185→  if (!open) return null;
   186→
   187→  function handleLogin() {
   188→    if (!password.trim()) {
   189→      toast.error("Enter password");
   190→      return;
   191→    }
   192→    setLoading(true);
   193→    setTimeout(() => {
   194→      const ok = admin.login("", password);
   195→      setLoading(false);
   196→      if (ok) {
   197→        toast.success("Admin login successful");
   198→      } else {
   199→        toast.error("Invalid password");
   200→      }
   201→    }, 400);
   202→  }
   203→
   204→  return (
   205→    <AnimatePresence>
   206→      {open && (
   207→        <motion.div
   208→          initial={{ opacity: 0 }}
   209→          animate={{ opacity: 1 }}
   210→          exit={{ opacity: 0 }}
   211→          className="fixed inset-0 z-[90] bg-black/80 backdrop-blur-md flex items-center justify-center p-4"
   212→          onClick={onClose}
   213→        >
   214→          <motion.div
   215→            initial={{ scale: 0.9, opacity: 0 }}
   216→            animate={{ scale: 1, opacity: 1 }}
   217→            exit={{ scale: 0.9, opacity: 0 }}
   218→            className="bg-slate-900 w-full max-w-sm rounded-2xl border border-violet-800/40 shadow-2xl overflow-hidden"
   219→            onClick={(e) => e.stopPropagation()}
   220→          >
   221→            {/* Header */}
   222→            <div className="bg-gradient-to-br from-violet-900 via-purple-900 to-fuchsia-900 text-white p-5 text-center relative overflow-hidden">
   223→              <div className="absolute -top-8 -right-8 w-32 h-32 bg-white/10 rounded-full" />
   224→              <div className="absolute -bottom-8 -left-8 w-32 h-32 bg-white/5 rounded-full" />

   226→                initial={{ scale: 0 }}

   228→                transition={{ type: "spring", stiffness: 200 }}

   231→                <Shield className="w-7 h-7" />
   232→              </motion.div>
   233→              <h2 className="font-black text-lg">APNA CRICKET ADMIN</h2>
   234→              <p className="text-[11px] text-white/70 mt-0.5">Restricted access · Password protected</p>
   235→            </div>
   236→
   237→            {/* Form — password only */}
   238→            <div className="p-5 space-y-3">

   241→                <div className="relative">
   242→                  <Input
   243→                    type={show ? "text" : "password"}
   244→                    value={password}
   245→                    onChange={(e) => setPassword(e.target.value)}
   246→                    placeholder="Enter password"
   247→                    className="bg-slate-800 border-slate-700 text-white placeholder:text-slate-500 h-12 text-base pr-10"

   268→                Close
   269→              </Button>

   272→        </motion.div>

   274→    </AnimatePresence>

   278→/* ---------- Dashboard Tab ---------- */
   279→
   280→function DashboardTab() {
   281→  // Select raw data and compute stats in useMemo to avoid infinite loop
   282→  const keys = useAdmin((s) => s.keys);
   283→  const devices = useAdmin((s) => s.devices);
   284→  const users = useAdmin((s) => s.users);
   285→  const logs = useAdmin((s) => s.logs);
   286→  const [syncing, setSyncing] = useState(false);
   287→
   288→  // Sync keys from Firebase — ensures all admin panels show SAME keys
   289→  async function syncFromFirebase() {
   290→    setSyncing(true);
   291→    try {
   292→      const { db } = await import("@/lib/firebase");
   293→      if (!db) {
   294→        toast.error("Firebase not configured");
   295→        return;
   296→      }
   297→      const { collection, getDocs } = await import("firebase/firestore");
   298→      const { syncKeyToFirestore, bulkSyncKeysToFirestore } = await import("@/lib/firebase-license");
   299→
   300→      // 1. Fetch all keys from Firebase
   301→      const snap = await getDocs(collection(db, "license_keys"));
   302→      const firebaseKeys: any[] = [];
   303→      snap.forEach((docSnap) => {
   304→        firebaseKeys.push({ key: docSnap.id, ...docSnap.data() });
   305→      });
   306→
   307→      // 2. Get local keys
   308→      const localKeys = useAdmin.getState().keys;
   309→      const firebaseKeySet = new Set(firebaseKeys.map((k: any) => k.key));
   310→
   311→      // 3. Sync local-only keys UP to Firebase
   312→      const localOnlyKeys = localKeys.filter((k) => !firebaseKeySet.has(k.key));
   313→      if (localOnlyKeys.length > 0) {
   314→        await bulkSyncKeysToFirestore(localOnlyKeys);
   315→      }
   316→
   317→      // 4. Merge Firebase keys DOWN to local
   318→      const localKeyMap = new Map(localKeys.map((k) => [k.key, k]));
   319→      const mergedKeys = [
   320→        ...localKeys,
   321→        ...firebaseKeys.filter((fk) => !localKeyMap.has(fk.key)).map((fk) => ({
   322→          key: fk.key,
   323→          type: fk.type || "trial",
   324→          status: fk.status || "active",
   325→          createdAt: fk.createdAt || Date.now(),
   326→          expiresAt: fk.expiresAt || null,
   327→          deviceIds: fk.deviceIds || [],
   328→          verifications: fk.verifications || 0,
   329→          userName: fk.userName || fk.mobile || "",
   330→          mobile: fk.mobile || "",
   331→        })),
   332→      ];
   333→
   334→      useAdmin.setState({ keys: mergedKeys });
   335→      toast.success(`✓ Synced: ${localKeys.length} local + ${firebaseKeys.length} Firebase = ${mergedKeys.length} total`);
   336→    } catch (err: any) {
   337→      toast.error(err?.message || "Sync failed");
   338→    } finally {
   339→      setSyncing(false);
   340→    }
   341→  }
   342→
   343→  const stats = (() => {
   344→    const now = Date.now();
   345→    const todayStart = new Date().setHours(0, 0, 0, 0);
   346→    return {
   347→      totalKeys: keys.length,
   348→      activeKeys: keys.filter((k) => k.status === "active" && (!k.expiresAt || k.expiresAt > now)).length,
   349→      usedKeys: keys.filter((k) => k.status === "used").length,
   350→      expiredKeys: keys.filter((k) => k.status === "expired" || (k.expiresAt && k.expiresAt < now)).length,
   351→      activeDevices: devices.filter((d) => d.status === "allowed").length,
   352→      todayVerifications: logs.filter((l) => l.type === "key_verification" && l.timestamp >= todayStart).length,
   353→      totalUsers: users.length,
   354→      teamsGeneratedToday: logs.filter((l) => l.type === "team_generation" && l.timestamp >= todayStart).length,
   355→    };
   356→  })();
   357→
   358→  const cards = [
   359→    { label: "Total Keys", value: stats.totalKeys, icon: Key, color: "from-violet-500 to-purple-600" },
   360→    { label: "Active Keys", value: stats.activeKeys, icon: ShieldCheck, color: "from-emerald-500 to-green-600" },
   361→    { label: "Used Keys", value: stats.usedKeys, icon: Check, color: "from-sky-500 to-blue-600" },
   362→    { label: "Expired Keys", value: stats.expiredKeys, icon: AlertTriangle, color: "from-amber-500 to-orange-600" },
   363→    { label: "Active Devices", value: stats.activeDevices, icon: Smartphone, color: "from-pink-500 to-rose-600" },
   364→    { label: "Today Verifications", value: stats.todayVerifications, icon: Activity, color: "from-cyan-500 to-teal-600" },
   365→    { label: "Total Users", value: stats.totalUsers, icon: Users, color: "from-indigo-500 to-violet-600" },
   366→    { label: "Teams Today", value: stats.teamsGeneratedToday, icon: Zap, color: "from-fuchsia-500 to-pink-600" },
   367→  ];
   368→  return (
   369→    <div className="space-y-4">
   370→      <div className="flex items-center justify-between">
   371→        <h3 className="text-sm font-bold text-white">Dashboard Overview</h3>
   372→        <Button
   373→          size="sm"
   374→          className="bg-emerald-600 hover:bg-emerald-700 text-white h-8 text-xs"
   375→          onClick={syncFromFirebase}
   376→          disabled={syncing}
   377→        >
   378→          {syncing ? <Loader2 className="w-3 h-3 mr-1 animate-spin" /> : <RefreshCw className="w-3 h-3 mr-1" />}
   379→          Sync from Firebase
   380→        </Button>
   381→      </div>
   382→      <div className="bg-emerald-900/20 border border-emerald-800/50 rounded-lg p-2 text-[10px] text-emerald-300">
   383→        ✓ All admin panels now sync from Firebase — keys, plans, and data are the SAME across all devices/APKs
   384→      </div>
   385→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   386→        {cards.map((c) => {
   387→          const Icon = c.icon;
   388→          return (
   389→            <div key={c.label} className="bg-slate-800 rounded-xl p-3 border border-slate-700">
   390→              <div className={cn("w-8 h-8 rounded-lg bg-gradient-to-br flex items-center justify-center mb-2", c.color)}>
   391→                <Icon className="w-4 h-4 text-white" />
   392→              </div>
   393→              <p className="text-2xl font-black text-white">{c.value}</p>
   394→              <p className="text-[10px] text-slate-400 font-semibold uppercase tracking-wide">{c.label}</p>
   395→            </div>
   396→          );
   397→        })}
   398→      </div>
   399→    </div>
   400→  );
   401→}
