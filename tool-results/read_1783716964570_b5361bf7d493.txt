    15→/**
    16→ * Strictly check if an error message indicates a CONFIRMED token expiry.
    17→ */
    18→function isConfirmedTokenExpiry(errorMsg: string, data: Record<string, unknown>): boolean {
    19→  const lowerMsg = errorMsg.toLowerCase().trim();
    20→
    21→  const confirmedTokenErrors = [
    22→    "invalid token",
    23→    "token expired",
    24→    "token is expired",
    25→    "access token expired",
    26→    "jwt expired",
    27→    "jwt malformed",
    28→    "invalid jwt",
    29→    "authentication failed",
    30→    "auth token invalid",
    31→    "login required",
    32→    "user not authenticated",
    33→    "not authenticated",
    34→    "session expired",
    35→    "account locked",
    36→    "expired token",
    37→    "invalid or expired",
    38→    "proxy returned 400",  // This means the actual platform API returned 400 = token expired
    39→    "proxy returned 401",  // 401 from platform = token expired
    40→    "proxy returned 403",  // 403 from platform = forbidden/expired
    41→  ];
    42→
    43→  for (const confirmed of confirmedTokenErrors) {
    44→    if (lowerMsg.includes(confirmed)) return true;
    45→  }
    46→
    47→  if (data.validToken === false && data.status === "fail") return true;
    48→
    49→  return false;
    50→}

    52→/**
    53→ * Determine if the response data indicates a successful transfer.
    54→ *
    55→ * The tgsoftware-api.online backend proxy has a KNOWN BUG:
    56→ * - It returns HTTP 201 + {status: "fail", message: "Something Went Wrong!"}
    57→ *   even when the team WAS actually created on the platform.
    58→ * - This happens because the proxy successfully forwards the request,
    59→ *   the platform creates the team, but the proxy fails to read the
    60→ *   response properly and returns its own error.
    61→ *
    62→ * Transfer result categories:
    63→ * 1. DEFINITE SUCCESS: HTTP 200 + {status: "success"}
    64→ * 2. TENTATIVE SUCCESS: HTTP 201 + {status: "fail", message: "Something Went Wrong!"}
    65→ *    → The team was likely created despite the error message
    66→ *    → We treat this as success (matching original APK behavior)
    67→ * 3. DEFINITE FAILURE: HTTP 201 + {status: "fail", message: "Proxy returned 400/401/403"}
    68→ *    → The platform API rejected the request (token expired)
    69→ * 4. OTHER FAILURES: HTTP 4xx/5xx or other error messages
    70→ *    → Treat as genuine failures
    71→ */
    72→function isTransferSuccess(httpStatus: number, data: Record<string, unknown>): {
    73→  success: boolean;
    74→  confidence: 'definite' | 'tentative';
    75→} {
    76→  // Method 1: Explicit success status field
    77→  if (data.status === "success") {
    78→    return { success: true, confidence: 'definite' };
    79→  }
    80→
    81→  // Method 2: HTTP 2xx + no explicit failure + success indicators
    82→  if (httpStatus >= 200 && httpStatus < 300 && data.status !== "fail" && data.status !== "error") {
    83→    const msg = String(data.message || "").toLowerCase();
    84→    const hasSuccessIndicator =
    85→      msg.includes("success") ||
    86→      msg.includes("created") ||
    87→      msg.includes("added") ||
    88→      msg.includes("updated") ||
    89→      msg.includes("saved") ||
    90→      data.data !== undefined;
    91→
    92→    if (hasSuccessIndicator) {
    93→      return { success: true, confidence: 'definite' };
    94→    }
    95→  }
    96→
    97→  // Method 3: HTTP 201 + {status: "fail"} with "Something Went Wrong!"
    98→  // This is the KNOWN BUG in the tgsoftware-api.online backend proxy.
    99→  // The team WAS actually created on the platform despite the error.
   100→  // We must treat this as a TENTATIVE SUCCESS (matching original APK behavior).
   101→  if (httpStatus === 201 && (data.status === "fail" || data.status === "error")) {
   102→    const msg = String(data.message || "").toLowerCase();
   103→
   104→    // Check if it's a genuine token/auth failure from the proxy
   105→    if (
   106→      msg.includes("proxy returned 400") ||
   107→      msg.includes("proxy returned 401") ||
   108→      msg.includes("proxy returned 403") ||
   109→      msg.includes("invalid token") ||
   110→      msg.includes("token expired") ||
   111→      msg.includes("jwt expired") ||
   112→      msg.includes("session expired") ||
   113→      msg.includes("not authenticated") ||
   114→      msg.includes("authentication failed")
   115→    ) {
   116→      // Genuine auth failure - the platform rejected the request
   117→      console.log(`[Transfer][AUTH_FAILURE] HTTP 201 + genuine auth error: ${data.message}. Treating as FAILURE.`);
   118→      return { success: false, confidence: 'definite' };
   119→    }
   120→
   121→    // "Something Went Wrong!" or other generic errors from the proxy
   122→    // These are false negatives - the team was likely created
   123→    console.log(`[Transfer][TENTATIVE_SUCCESS] HTTP 201 + body status=${data.status}, message="${data.message}". Backend proxy bug - treating as TENTATIVE SUCCESS (team likely created).`);
   124→    return { success: true, confidence: 'tentative' };
   125→  }
   126→
   127→  // All other cases: failure
   128→  return { success: false, confidence: 'definite' };
   129→}

   131→/**
   132→ * Validate all required transfer parameters before sending request.
   133→ */
   134→function validateTransferParams(params: {
   135→  authToken?: string;
   136→  matchId?: string | number;
   137→  captain?: number | string;
   138→  vice_captain?: number | string;
   139→  players?: number[] | unknown;
   140→  fantasyApp?: string;
   141→  sportIndex?: number;
   142→  type?: string;
   143→  id?: string;
   144→}): { valid: boolean; error?: string; code?: string } {
   145→  if (!params.authToken) {
   146→    return { valid: false, error: `Account not linked. Please link your ${params.fantasyApp || "fantasy"} account first via OTP verification.`, code: "NO_AUTH_TOKEN" };
   147→  }
   148→  if (!params.matchId) {
   149→    return { valid: false, error: "Match ID is required for transfer", code: "MISSING_MATCH_ID" };
   150→  }
   151→  if (!params.captain || params.captain === 0) {
   152→    return { valid: false, error: "Captain ID is required (fantasy player ID)", code: "MISSING_CAPTAIN" };
   153→  }
   154→  if (!params.vice_captain || params.vice_captain === 0) {
   155→    return { valid: false, error: "Vice-Captain ID is required (fantasy player ID)", code: "MISSING_VICE_CAPTAIN" };
   156→  }
   157→  if (!params.players || !Array.isArray(params.players) || params.players.length === 0) {
   158→    return { valid: false, error: "Team players are required (array of fantasy IDs)", code: "MISSING_PLAYERS" };
   159→  }
   160→  if (params.players.length < 11) {
   161→    return { valid: false, error: `Team must have at least 11 players, got ${params.players.length}. Missing fantasy IDs for this platform.`, code: "INSUFFICIENT_PLAYERS" };
   162→  }
   163→  if (!params.fantasyApp) {
   164→    return { valid: false, error: "Fantasy platform is required", code: "MISSING_FANTASY_APP" };
   165→  }
   166→  // FIX: If type is "edit" but no team ID, auto-convert to "new" (add) instead of failing.
   167→  // This happens when list-of-teams returned empty (no existing teams on platform).
   168→  if (params.type === "edit" && !params.id) {
   169→    console.log("[Transfer][VALIDATE] Edit requested but no team ID — converting to add (new)");
   170→    params.type = "new";
   171→  }
   172→  return { valid: true };
   173→}

   243→function toNumber(val: unknown): number | null {
   244→  if (typeof val === "number" && !isNaN(val)) return val;
   245→  if (typeof val === "string") {

   252→/**
   253→ * Pre-verify auth token before transfer to avoid wasting time on expired tokens.
   254→ * Returns true if token appears valid, false if confirmed expired, null if can't verify.
   255→ */
   256→async function preVerifyToken(authToken: string, fantasyApp: string, matchId?: string): Promise<boolean | null> {
   257→  try {
   258→    const res = await fetch(`${BACKEND_URL}/api/fantasy/auth/verify`, {
   259→      method: "POST",
   260→      headers: { "Content-Type": "application/json" },
   261→      body: JSON.stringify({ fantasyApp, authToken, matchId: matchId || "" }),
   262→      signal: AbortSignal.timeout(8000),
   263→    });
   264→
   265→    const responseText = await res.text();
   266→    let data: Record<string, unknown>;
   267→    try {
   268→      data = JSON.parse(responseText);
   269→    } catch {
   270→      // Can't parse - can't verify, don't block
   271→      return null;
   272→    }
   273→
   274→    // If backend confirms token is invalid
   275→    if (data.validToken === false) return false;
   276→    if (res.status === 401 || res.status === 403) return false;
   277→
   278→    // If backend confirms token is valid
   279→    if (data.validToken === true) return true;
   280→
   281→    // Update token if backend provides a new one
   282→    if (data.updateUiToken && data.authToken) {
   283→      // Return the updated token info - caller can use this
   284→      return true;
   285→    }
   286→
   287→    // Can't determine - don't block
   288→    return null;
   289→  } catch {
   290→    // Network error - don't block
   291→    return null;
   292→  }
   293→}

   539→          if (typeof data.data === "string" && data.data) {
   540→            const decrypted = decryptAES(data.data);
   541→            if (decrypted) {
   542→              try { responseData = JSON.parse(decrypted); } catch { responseData = { rawDecrypted: decrypted }; }
   543→            }
   544→          }
   545→
   546→          console.log(`[Transfer][SUCCESS] Team transferred to ${app} (${operation}) - HTTP ${res.status} (confidence: ${transferResult.confidence})`);
   547→
   548→          return NextResponse.json({
   549→            status: "success",
   550→            method: endpointUrl.includes("classic") ? "classic" : "fantasy",
   551→            endpoint: endpointUrl,
   552→            platform: app,
   553→            operation: operation,
   554→            data: responseData || null,
   555→            httpStatus: res.status,
   556→            confidence: transferResult.confidence,
   557→            message: transferResult.confidence === 'tentative'
   558→              ? `Team ${operation === "edit" ? "replaced" : "added"} on ${app} (verified by proxy)`
   559→              : `Team ${operation === "edit" ? "replaced" : "added"} on ${app} successfully`,
   560→          }, { status: 200 });
   561→        }
   562→
   563→        // Handle failure
   564→        lastError = (data.message as string) || (data.error as string) || `HTTP ${res.status}`;
   565→        lastData = data;
   566→
   567→        console.error(`[Transfer][FAILED] ${lastError} (HTTP ${res.status})`);
   568→
   569→        // If it's a confirmed token expiry, stop immediately
   570→        if (isConfirmedTokenExpiry(lastError, data)) {
   571→          console.error(`[Transfer][AUTH] Token expired confirmed. Stopping.`);
   572→          return NextResponse.json({
   573→            status: "fail",
   574→            error: `Session expired on ${app}. Please re-link your account via OTP verification.`,
   575→            code: "TOKEN_EXPIRED",
   576→            needsReauth: true,
   577→            backendError: lastError,
   578→          }, { status: 200 });
   579→        }
   580→
   581→        // Check for deadline passed
   582→        const lowerMsg = lastError.toLowerCase();
   583→        if (lowerMsg.includes("deadline") || lowerMsg.includes("match expired") || lowerMsg.includes("match started")) {
   584→          return NextResponse.json({
   585→            status: "fail",
   586→            error: "Match deadline has passed. You can no longer create or edit teams for this match.",
   587→            code: "DEADLINE_PASSED",
   588→            backendError: lastError,
   589→          }, { status: 200 });
   590→        }
   591→
   592→        // Check for team limit reached
   593→        if (lowerMsg.includes("limit") || lowerMsg.includes("maximum") || lowerMsg.includes("already")) {
   594→          return NextResponse.json({
   595→            status: "fail",
   596→            error: `Team limit reached on ${app}. You already have the maximum number of teams for this match.`,
   597→            code: "TEAM_LIMIT_REACHED",
   598→            backendError: lastError,
   599→          }, { status: 200 });
   600→        }
   601→
   602→        // For 404, skip to next endpoint
   603→        if (res.status === 404) {
   604→          console.log(`[Transfer][SKIP] Endpoint returned 404. Trying next...`);
   605→          continue;
   606→        }
   607→
   608→        // For other errors, try next endpoint
   609→        continue;
   610→
   611→      } catch (fetchErr) {
   612→        const errMsg = fetchErr instanceof Error ? fetchErr.message : String(fetchErr);
   613→        console.error(`[Transfer][NETWORK] Fetch error: ${errMsg}`);
   614→        lastError = errMsg.includes("timeout") ? "Transfer request timed out (30s)" :
   615→                    errMsg.includes("abort") ? "Request was aborted" :
   616→                    `Network error: ${errMsg}`;
   617→        lastHttpStatus = 0;
   618→        continue;
   619→      }
   620→    }
   621→
   622→    // If edit failed, try add as fallback
   623→    if (operation === "edit" && lastError && !isConfirmedTokenExpiry(lastError, lastData || {})) {
   624→      console.log(`[Transfer][FALLBACK] Edit failed on all endpoints. Trying add as fallback...`);
   625→
   626→      const fallbackPayload = { ...payload };
   627→      delete fallbackPayload.id;
   628→      fallbackPayload.type = "new";
   629→
   630→      const addEndpoints = config.addEndpoints;
   631→
   632→      for (const fallbackUrl of addEndpoints) {
   633→        console.log(`[Transfer][FALLBACK] Trying add endpoint: ${fallbackUrl}`);
   634→
   635→        try {
   636→          const fallbackStart = Date.now();
   637→          const fallbackRes = await fetch(fallbackUrl, {
   638→            method: "POST",
   639→            headers: { "Content-Type": "application/json" },
   640→            body: JSON.stringify(fallbackPayload),
   641→            signal: AbortSignal.timeout(30000),
   642→          });
   643→          const fallbackDuration = Date.now() - fallbackStart;
   644→
   645→          const fallbackText = await fallbackRes.text();
   646→
   647→          console.log(`[Transfer][FALLBACK] HTTP ${fallbackRes.status}, duration: ${fallbackDuration}ms`);
   648→          console.log(`[Transfer][FALLBACK] Response: ${fallbackText.substring(0, 500)}`);
   649→
   650→          let fallbackData: Record<string, unknown>;
   651→          try {
   652→            fallbackData = JSON.parse(fallbackText);
   653→          } catch {
   654→            console.error(`[Transfer][FALLBACK] Non-JSON response`);
   655→            fallbackData = {};
   656→          }
   657→
   658→          // Use improved success check
   659→          const fallbackCheck = isTransferSuccess(fallbackRes.status, fallbackData);
   660→          if (fallbackCheck.success) {
   661→            let responseData = fallbackData.data;
   662→            if (typeof fallbackData.data === "string" && fallbackData.data) {
   663→              const decrypted = decryptAES(fallbackData.data);
   664→              if (decrypted) {
   665→                try { responseData = JSON.parse(decrypted); } catch { responseData = { rawDecrypted: decrypted }; }
   666→              }
   667→            }
   668→
   669→            console.log(`[Transfer][FALLBACK][SUCCESS] Team added via fallback (confidence: ${fallbackCheck.confidence})`);
   670→
   671→            return NextResponse.json({
   672→              status: "success",
   673→              method: "fallback_add",
   674→              endpoint: fallbackUrl,
   675→              platform: app,
   676→              operation: "add",
   677→              data: responseData || null,
   678→              httpStatus: fallbackRes.status,
   679→              confidence: fallbackCheck.confidence,
   680→              message: fallbackCheck.confidence === 'tentative'
   681→                ? `Team added to ${app} (verified by proxy, fallback)`
   682→                : `Team added to ${app} successfully (fallback from edit)`,
   683→            }, { status: 200 });
   684→          }
   685→
   686→          const fallbackError = (fallbackData?.message as string) || (fallbackData?.error as string) || "Unknown fallback error";
   687→          console.error(`[Transfer][FALLBACK][FAILED] ${fallbackError}`);
   688→
   689→          // Check for token expiry in fallback too
   690→          if (isConfirmedTokenExpiry(fallbackError, fallbackData)) {
   691→            return NextResponse.json({
   692→              status: "fail",
   693→              error: `Session expired on ${app}. Please re-link your account via OTP verification.`,
   694→              code: "TOKEN_EXPIRED",
   695→              needsReauth: true,
   696→              backendError: fallbackError,
   697→            }, { status: 200 });
   698→          }
   699→
   700→          if (fallbackError && fallbackError !== "Something Went Wrong!") {
   701→            lastError = fallbackError;
   702→          }
   703→          lastHttpStatus = fallbackRes.status;
   704→        } catch (fallbackErr) {
   705→          console.error(`[Transfer][FALLBACK][ERROR] ${fallbackErr instanceof Error ? fallbackErr.message : "unknown"}`);
   706→        }
   707→      }
   708→    }
   709→
   710→    // Return detailed error
   711→    console.error(`\n${"!".repeat(60)}`);
   712→    console.error(`[Transfer][FINAL] ALL ENDPOINTS FAILED for ${app} ${operation}`);
   713→    console.error(`[Transfer][FINAL] Backend error: ${lastError}`);
   714→    console.error(`[Transfer][FINAL] HTTP status: ${lastHttpStatus}`);
   715→    console.error(`[Transfer][FINAL] Endpoints tried: ${endpointChain.join(", ")}`);
   716→    console.error(`[Transfer][FINAL] Token: len=${preparedToken.length}, isJWT=${preparedToken.startsWith("eyJ")}, isJSON=${preparedToken.startsWith("{")}`);
   717→    console.error(`${"!".repeat(60)}\n`);
   718→
   719→    let displayError = lastError || "Unknown transfer error";
   720→
   721→    if (displayError === "All proxy attempts failed" || displayError.includes("proxy attempts failed")) {
   722→      displayError = `The ${app} server rejected the transfer. Your session may have expired. Please re-link your account via OTP verification and try again.`;
   723→    } else if (displayError === "Something Went Wrong!") {
   724→      displayError = `The ${app} server returned an error. Your session has likely expired - please re-link your account via OTP verification.`;
   725→    } else if (displayError === "Proxy returned 400") {
   726→      displayError = `Your ${app} session has expired (the platform rejected the request). Please re-link your account via OTP verification.`;
   727→    } else if (displayError === "Error while transfering the team!") {
   728→      displayError = `Failed to transfer team on ${app}. Your auth session has likely expired. Try re-linking your ${app} account via OTP verification.`;
   729→    } else if (displayError.includes("non-JSON response") && lastHttpStatus === 0) {
   730→      displayError = `Could not reach the ${app} transfer server. Please check your internet connection and try again.`;
   731→    }
   732→
   733→    return NextResponse.json({
   734→      status: "fail",
   735→      error: displayError,
   736→      code: "TRANSFER_ERROR",
   737→      needsReauth: false,
   738→      backendError: lastError,
   739→      httpStatus: lastHttpStatus,
   740→      platform: app,
   741→      operation: operation,
   742→      endpointsTried: endpointChain,
   743→      details: {
   744→        matchId: String(matchId),
   745→        captain: finalCaptain,
   746→        vice_captain: finalViceCaptain,
   747→        playerCount: numericPlayerIds.length,
   748→        teamId: id || null,
   749→        tokenLength: preparedToken.length,
   750→        tokenIsJWT: preparedToken.startsWith("eyJ"),
   751→        tokenIsJSON: preparedToken.startsWith("{"),
   752→      },
   753→    }, { status: 200 });
   754→
   755→  } catch (error) {
   756→    console.error("[Transfer][ROUTE_ERROR]", error);
   757→    return NextResponse.json({
   758→      status: "fail",
   759→      error: `Server error: ${error instanceof Error ? error.message : "Unknown error"}`,
   760→      code: "SERVER_ERROR",
   761→      needsReauth: false,
   762→    }, { status: 200 });
   763→  }
   764→}
