     4→ * Hidden admin mode with license management, device binding, user control,

   187→function randomKeySegment(): string {
   188→  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
   189→  let out = "";
   190→  for (let i = 0; i < 4; i++) {
   191→    out += chars[Math.floor(Math.random() * chars.length)];
   192→  }
   193→  return out;
   194→}

   196→function generateLicenseKey(prefix: string = LICENSE_PREFIX_DEFAULT): string {
   197→  return `${prefix}-${randomKeySegment()}-${randomKeySegment()}-${randomKeySegment()}`;
   198→}

   200→export function getExpiryForType(type: KeyType): number | null {
   201→  const now = Date.now();
   202→  const day = 86400000;
   203→  const hour = 3600000;
   204→  switch (type) {
   205→    case "trial": return now + 1 * day;        // 1 day free
   206→    case "match_basic": return now + 6 * hour; // 1 match = 6 hours
   207→    case "match_full": return now + 1 * day;   // 24 hour active
   208→    case "weekly": return now + 7 * day;       // 7 days
   209→    case "monthly": return now + 30 * day;     // 30 days (Pro ₹299)
   210→    case "elite": return now + 90 * day;       // 90 days (Elite ₹999)
   211→    case "lifetime": return null;              // forever
   212→  }
   213→}

  1060→const LICENSE_STORAGE_KEY = "apna-cricket-license";

  1062→export interface LicenseState {
  1063→  verified: boolean;
  1064→  key: string | null;
  1065→  deviceId: string;
  1066→  keyType?: KeyType | null; // "trial" | "weekly" | "monthly" | "elite" | "lifetime"
  1067→}
  1068→
  1069→export function getDeviceId(): string {
  1070→  if (typeof window === "undefined") return "server";
  1071→  let id = localStorage.getItem("apna-cricket-device-id");
  1072→  if (!id) {
  1073→    id = "dev_" + Math.random().toString(36).slice(2, 12) + Date.now().toString(36);
  1074→    localStorage.setItem("apna-cricket-device-id", id);
  1075→  }
  1076→  return id;
  1077→}
  1078→
  1079→export function getLicenseState(): LicenseState {
  1080→  if (typeof window === "undefined") return { verified: false, key: null, deviceId: "server", keyType: null };
  1081→  try {
  1082→    const data = JSON.parse(localStorage.getItem(LICENSE_STORAGE_KEY) || "{}");
  1083→    return {
  1084→      verified: !!data.verified,
  1085→      key: data.key || null,
  1086→      deviceId: getDeviceId(),
  1087→      keyType: data.keyType || null,
  1088→    };
  1089→  } catch {
  1090→    return { verified: false, key: null, deviceId: getDeviceId(), keyType: null };
  1091→  }
  1092→}
  1093→
  1094→export function setLicenseVerified(key: string, keyType?: KeyType | null) {
  1095→  if (typeof window === "undefined") return;
  1096→  localStorage.setItem(
  1097→    LICENSE_STORAGE_KEY,
  1098→    JSON.stringify({
  1099→      verified: true,
  1100→      key,
  1101→      deviceId: getDeviceId(),
  1102→      keyType: keyType || null,
  1103→    })
  1104→  );
  1105→}
  1106→
  1107→export function clearLicense() {
  1108→  if (typeof window === "undefined") return;
  1109→  localStorage.removeItem(LICENSE_STORAGE_KEY);
  1110→}
