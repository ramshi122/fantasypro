   260→  // Auto-replace non-playing players in the given teams
   261→  function handleAutoReplaceAll(teamsToFix: TeamRow[]) {
   262→    try {
   263→      const all = JSON.parse(localStorage.getItem("apna-cricket-local-teams") || "{}");
   264→      const list = all[matchId] || [];
   265→      let totalReplaced = 0;
   266→
   267→      for (const team of teamsToFix) {
   268→        const idx = list.findIndex(
   269→          (t: TeamRow) => t._id === team._id || t.teamId === team.teamId
   270→        );
   271→        if (idx === -1) continue;
   272→
   273→        const { updatedTeam, replaced } = autoReplaceNonPlaying(
   274→          list[idx],
   275→          players as MatchPlayer[],
   276→          true
   277→        );
   278→        list[idx] = updatedTeam;
   279→        totalReplaced += replaced.length;
   280→      }
   281→
   282→      all[matchId] = list;
   283→      localStorage.setItem("apna-cricket-local-teams", JSON.stringify(all));
   284→
   285→      if (totalReplaced > 0) {
   286→        toast.success(`Replaced ${totalReplaced} non-playing player(s) across ${teamsToFix.length} team(s)`);
   287→      } else {
   288→        toast.info("No replacements available — all non-playing players kept (insufficient playing XI)");
   289→      }
   290→      setRefreshKey((k) => k + 1);
   291→    } catch (err: any) {
   292→      toast.error("Failed to auto-replace players");
   293→    }
   294→  }

   299→  function handleBulkAddTeams(teams: TeamRow[]) {

   309→      const key = String(matchId);
   310→
   311→      // Read existing teams (to count how many were wiped)
   312→      let all: Record<string, TeamRow[]> = {};
   313→      try {
   314→        all = JSON.parse(localStorage.getItem(LOCAL_TEAMS_KEY) || "{}");
   315→      } catch {
   316→        all = {};
   317→      }
   318→
   319→      const oldCount = (all[key] || []).length;
   320→
   321→      // WIPE old teams — only new teams will be saved
   322→      const newTeams: TeamRow[] = teams.map((t, i) => ({
   323→        _id: "local_" + Math.random().toString(36).slice(2, 10) + Date.now().toString(36),
   324→        teamId: "local_" + Math.random().toString(36).slice(2, 10),
   325→        teamNo: i + 1,
   326→        team_number: i + 1,
   327→        players: t.players || [],
   328→        captain: t.captain,
   329→        viceCaptain: t.viceCaptain,
   330→        vicecaptain: t.viceCaptain,
   331→        c: t.captain,
   332→        vc: t.viceCaptain,
   333→        team: t.team,
   334→        type: t.type ?? "sweep",
   335→        transferStatus: "pending",
   336→      }));
   337→
   338→      // Save: ONLY new teams (old teams wiped)
   339→      all[key] = newTeams;
   340→      localStorage.setItem(LOCAL_TEAMS_KEY, JSON.stringify(all));
   341→
   342→      // VERIFY save worked
   343→      const verifyData = JSON.parse(localStorage.getItem(LOCAL_TEAMS_KEY) || "{}");
   344→      const verifyTeams = verifyData[key] || [];
   345→      console.log(`[handleBulkAddTeams] Wiped ${oldCount} old. Saved ${newTeams.length}. Verified: ${verifyTeams.length} in localStorage.`);
   346→
   347→      // IMMEDIATELY update the state (don't wait for refreshKey)
   348→      setExistingTeams([...newTeams]);
   349→
   350→      // Force a complete reload after a short delay to ensure UI updates
   351→      setTimeout(() => {
   352→        setRefreshKey((k) => k + 1);
   353→      }, 100);
   354→
   355→      if (oldCount > 0) {
   356→        toast.success(`✓ Wiped ${oldCount} old teams. Generated ${newTeams.length} new teams!`);
   357→      } else {
   358→        toast.success(`✓ Generated ${newTeams.length} new teams!`);
   359→      }
   360→    } catch (err) {
   361→      console.error("[handleBulkAddTeams] Save error:", err);
   362→      toast.error("Failed to save teams — try again");
   363→      return;
   364→    }
   365→
   366→    // === Integration: analytics + performance tracker + cashback + leaderboard ===
   367→    (async () => {
   368→      try {
   369→        const { trackEvent } = await import("@/lib/analytics");
   370→        const { trackMatchTeam } = await import("@/lib/performance-tracker");
   371→        const { processCashback } = await import("@/lib/cashback");
   372→        const { getDeviceId, getLicenseState } = await import("@/lib/admin-store");
   373→        trackEvent("team_generation", {
   374→          matchId,
   375→          teamCount: teams.length,
   376→          source: "sweep_generator",
   377→        });

   380→        // Cashback (₹0.50/team, max ₹10/day)

   385→            toast.success(`💰 ₹${result.cashback} cashback credited!`);
   386→          }
   387→        }
   388→
   389→        // Update leaderboard
   390→        try {
   391→          await fetch("/api/leaderboard", {
   392→            method: "POST",
   393→            headers: { "Content-Type": "application/json" },

   395→              action: "recordTeamGen",
   396→              userId: license.deviceId || getDeviceId(),
   397→              userName: license.key || "Guest",
   398→              count: teams.length,
   399→              matchId,
