     1→/**
     2→ * Admin Panel Store — LOCALSTORAGE IS THE PRIMARY STORE
     3→ * ------------------------------------------------------
     4→ * Hidden admin mode with license management, device binding, user control,
     5→ * team generator toggles, announcements, activity logs, and settings.
     6→ *
     7→ * Storage model (per FIREBASE-CLEANUP):
     8→ *   • localStorage  — primary store for keys/devices/users/logs/settings
     9→ *                     (admin-generated + 500+ seed keys; same on every phone
    10→ *                     because the seed generation is deterministic-ish and
    11→ *                     the same initial keys are seeded on every device)
    12→ *   • Firebase      — used ONLY for license key verification
    13→ *                     (single-doc reads of license_keys/{keyStr})
    14→ *
    15→ * When admin GENERATES a key → save to localStorage AND sync the ONE key
    16→ * to Firebase (setDoc by key string) + record in admin_generated_keys.
    17→ *
    18→ * When admin CHECKS keys → read from localStorage (seed + generated keys).
    19→ *
    20→ * When a USER verifies a key → read ONE doc from Firebase by key string
    21→ * (NOT a collection download) — handled in `verifyLicense` via
    22→ * `fetchKeyFromFirestore`.
    23→ *
    24→ * All real-time `onSnapshot` listeners have been REMOVED.
    25→ * One-shot reads/writes only.
    26→ */

   205→function randomKeySegment(): string {
   206→  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
   207→  let out = "";
   208→  for (let i = 0; i < 4; i++) {
   209→    out += chars[Math.floor(Math.random() * chars.length)];
   210→  }
   211→  return out;
   212→}

   214→function generateLicenseKey(prefix: string = LICENSE_PREFIX_DEFAULT): string {
   215→  return `${prefix}-${randomKeySegment()}-${randomKeySegment()}-${randomKeySegment()}`;
   216→}

   218→export function getExpiryForType(type: KeyType): number | null {
   219→  const now = Date.now();
   220→  const day = 86400000;
   221→  const hour = 3600000;
   222→  switch (type) {
   223→    case "trial": return now + 1 * day;          // 1 day free
   224→    case "match_basic": return now + 6 * hour;   // 1 match = 6 hours
   225→    case "match_full": return now + 1 * day;     // 24 hour active
   226→    case "weekly": return now + 7 * day;         // 7 days
   227→    case "monthly": return now + 30 * day;       // 30 days (Pro ₹299)
   228→    case "half_yearly": return now + 180 * day;  // 180 days (Semi-Annual ₹499)
   229→    case "elite": return now + 90 * day;         // 90 days (Elite ₹999)
   230→    case "lifetime": return null;                // forever
   231→  }
   232→}

   240→function lsGet<T>(key: string, fallback: T): T {
   241→  if (typeof window === "undefined") return fallback;
   242→  try {
   243→    const raw = localStorage.getItem(key);
   244→    if (!raw) return fallback;
   245→    return JSON.parse(raw) as T;
   246→  } catch {
   247→    return fallback;
   248→  }
   249→}
   250→
   251→function lsSet<T>(key: string, value: T): void {
   252→  if (typeof window === "undefined") return;
   253→  try {
   254→    localStorage.setItem(key, JSON.stringify(value));
   255→  } catch {
   256→    /* ignore quota errors */
   257→  }
   258→}

   308→/**
   309→ * Bootstrap initial state from localStorage. If localStorage is empty for
   310→ * a given collection (first run on this device), seed it with the initial
   311→ * demo data so that EVERY phone sees the same set of seed keys + devices +
   312→ * users + logs.
   313→ */
   314→function bootstrapInitialState() {
   315→  const initial = getInitialData();
   316→
   317→  const persistedKeys = lsGet<LicenseKey[] | null>(LS_KEYS, null);
   318→  const keys = persistedKeys && persistedKeys.length > 0 ? persistedKeys : initial.keys;
   319→  if (!persistedKeys || persistedKeys.length === 0) lsSet(LS_KEYS, keys);
   320→
   321→  const persistedDevices = lsGet<Device[] | null>(LS_DEVICES, null);
   322→  const devices = persistedDevices && persistedDevices.length > 0 ? persistedDevices : initial.devices;
   323→  if (!persistedDevices || persistedDevices.length === 0) lsSet(LS_DEVICES, devices);
   324→
   325→  const persistedUsers = lsGet<User[] | null>(LS_USERS, null);
   326→  const users = persistedUsers && persistedUsers.length > 0 ? persistedUsers : initial.users;
   327→  if (!persistedUsers || persistedUsers.length === 0) lsSet(LS_USERS, users);
   328→
   329→  const persistedLogs = lsGet<ActivityLog[] | null>(LS_LOGS, null);
   330→  const logs = persistedLogs && persistedLogs.length > 0 ? persistedLogs : initial.logs;
   331→  if (!persistedLogs || persistedLogs.length === 0) lsSet(LS_LOGS, logs);
   332→
   333→  const persistedSettings = lsGet<AdminSettings | null>(LS_SETTINGS, null);
   334→  const settings = persistedSettings ?? DEFAULT_SETTINGS;
   335→  if (!persistedSettings) lsSet(LS_SETTINGS, settings);
   336→
   337→  const announcements = lsGet<Announcement[]>(LS_ANNOUNCEMENTS, []);
   338→
   339→  return { keys, devices, users, logs, settings, announcements };
   340→}

  1019→export function getDeviceId(): string {
  1020→  if (typeof window === "undefined") return "server";
  1021→  let id = localStorage.getItem("apna-cricket-device-id");
  1022→  if (!id) {
  1023→    id = "dev_" + Math.random().toString(36).slice(2, 12) + Date.now().toString(36);
  1024→    localStorage.setItem("apna-cricket-device-id", id);
  1025→  }
  1026→  return id;

  1029→export function getLicenseState(): LicenseState {
  1030→  if (typeof window === "undefined") return { verified: false, key: null, deviceId: "server", keyType: null };
  1031→  try {
  1032→    const data = JSON.parse(localStorage.getItem(LICENSE_STORAGE_KEY) || "{}");
  1033→    return {
  1034→      verified: !!data.verified,
  1035→      key: data.key || null,
  1036→      deviceId: getDeviceId(),
  1037→      keyType: data.keyType || null,
  1038→    };
  1039→  } catch {
  1040→    return { verified: false, key: null, deviceId: getDeviceId(), keyType: null };
  1041→  }
