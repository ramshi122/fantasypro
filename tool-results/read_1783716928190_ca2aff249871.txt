     1→/**
     2→ * Admin Panel Store — LOCALSTORAGE IS THE PRIMARY STORE
     3→ * ------------------------------------------------------
     4→ * Hidden admin mode with license management, device binding, user control,
     5→ * team generator toggles, announcements, activity logs, and settings.
     6→ *
     7→ * Storage model (per FIREBASE-CLEANUP):
     8→ *   • localStorage  — primary store for keys/devices/users/logs/settings
     9→ *                     (admin-generated + 500+ seed keys; same on every phone
    10→ *                     because the seed generation is deterministic-ish and
    11→ *                     the same initial keys are seeded on every device)
    12→ *   • Firebase      — used ONLY for license key verification
    13→ *                     (single-doc reads of license_keys/{keyStr})
    14→ *
    15→ * When admin GENERATES a key → save to localStorage AND sync the ONE key
    16→ * to Firebase (setDoc by key string) + record in admin_generated_keys.
    17→ *
    18→ * When admin CHECKS keys → read from localStorage (seed + generated keys).
    19→ *
    20→ * When a USER verifies a key → read ONE doc from Firebase by key string
    21→ * (NOT a collection download) — handled in `verifyLicense` via
    22→ * `fetchKeyFromFirestore`.
    23→ *
    24→ * All real-time `onSnapshot` listeners have been REMOVED.
    25→ * One-shot reads/writes only.
    26→ */

   205→function randomKeySegment(): string {
   206→  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
   207→  let out = "";
   208→  for (let i = 0; i < 4; i++) {
   209→    out += chars[Math.floor(Math.random() * chars.length)];
   210→  }
   211→  return out;
   212→}

   218→export function getExpiryForType(type: KeyType): number | null {
   219→  const now = Date.now();
   220→  const day = 86400000;
   221→  const hour = 3600000;
   222→  switch (type) {
   223→    case "trial": return now + 1 * day;        // 1 day free
   224→    case "match_basic": return now + 6 * hour; // 1 match = 6 hours
   225→    case "match_full": return now + 1 * day;   // 24 hour active
   226→    case "weekly": return now + 7 * day;       // 7 days
   227→    case "monthly": return now + 30 * day;     // 30 days (Pro ₹299)
   228→    case "elite": return now + 90 * day;       // 90 days (Elite ₹999)
   229→    case "lifetime": return null;              // forever
   230→  }
   231→}

   239→function lsGet<T>(key: string, fallback: T): T {
   240→  if (typeof window === "undefined") return fallback;
   241→  try {
   242→    const raw = localStorage.getItem(key);
   243→    if (!raw) return fallback;
   244→    return JSON.parse(raw) as T;
   245→  } catch {
   246→    return fallback;
   247→  }
   248→}
   249→
   250→function lsSet<T>(key: string, value: T): void {
   251→  if (typeof window === "undefined") return;
   252→  try {
   253→    localStorage.setItem(key, JSON.stringify(value));
   254→  } catch {
   255→    /* ignore quota errors */
   256→  }
   257→}

   351→/**
   352→ * Bootstrap initial state from localStorage. If localStorage is empty for
   353→ * a given collection (first run on this device), seed it with the initial
   354→ * demo data so that EVERY phone sees the same set of seed keys + devices +
   355→ * users + logs.
   356→ */
   357→function bootstrapInitialState() {
   358→  const initial = getInitialData();
   359→
   360→  const persistedKeys = lsGet<LicenseKey[] | null>(LS_KEYS, null);
   361→  const keys = persistedKeys && persistedKeys.length > 0 ? persistedKeys : initial.keys;
   362→  if (!persistedKeys || persistedKeys.length === 0) lsSet(LS_KEYS, keys);
   363→
   364→  const persistedDevices = lsGet<Device[] | null>(LS_DEVICES, null);
   365→  const devices = persistedDevices && persistedDevices.length > 0 ? persistedDevices : initial.devices;
   366→  if (!persistedDevices || persistedDevices.length === 0) lsSet(LS_DEVICES, devices);
   367→
   368→  const persistedUsers = lsGet<User[] | null>(LS_USERS, null);
   369→  const users = persistedUsers && persistedUsers.length > 0 ? persistedUsers : initial.users;
   370→  if (!persistedUsers || persistedUsers.length === 0) lsSet(LS_USERS, users);
   371→
   372→  const persistedLogs = lsGet<ActivityLog[] | null>(LS_LOGS, null);
   373→  const logs = persistedLogs && persistedLogs.length > 0 ? persistedLogs : initial.logs;
   374→  if (!persistedLogs || persistedLogs.length === 0) lsSet(LS_LOGS, logs);
   375→
   376→  const persistedSettings = lsGet<AdminSettings | null>(LS_SETTINGS, null);
   377→  const settings = persistedSettings ?? DEFAULT_SETTINGS;
   378→  if (!persistedSettings) lsSet(LS_SETTINGS, settings);
   379→
   380→  const announcements = lsGet<Announcement[]>(LS_ANNOUNCEMENTS, []);
   381→
   382→  return { keys, devices, users, logs, settings, announcements };
   383→}

  1055→export function getLicenseState(): LicenseState {
  1056→  if (typeof window === "undefined") return { verified: false, key: null, deviceId: "server", keyType: null };
  1057→  try {
  1058→    const data = JSON.parse(localStorage.getItem(LICENSE_STORAGE_KEY) || "{}");
  1059→    return {
  1060→      verified: !!data.verified,
  1061→      key: data.key || null,
  1062→      deviceId: getDeviceId(),
  1063→      keyType: data.keyType || null,
  1064→    };
  1065→  } catch {
  1066→    return { verified: false, key: null, deviceId: getDeviceId(), keyType: null };
  1067→  }
  1068→}
