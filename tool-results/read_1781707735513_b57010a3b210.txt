    58→type Step = "loading" | "ready" | "saving" | "error";
    59→
    60→export function TransferFlow({ match }: { match: MatchInfo }) {
    61→  const token = useAuth((s) => s.token);
    62→  const isGuest = useAuth((s) => s.isGuest);
    63→  const license = useLicense();
    64→  const [step, setStep] = useState<Step>("loading");
    65→  const [existingTeams, setExistingTeams] = useState<TeamRow[]>([]);
    66→  const [players, setPlayers] = useState<MatchPlayer[]>([]);
    67→  const [error, setError] = useState<string | null>(null);
    68→  const [showEditor, setShowEditor] = useState(false);
    69→  const [editingTeam, setEditingTeam] = useState<TeamRow | null>(null);
    70→  const [showLinkModal, setShowLinkModal] = useState(false);
    71→  const [showSweepGenerator, setShowSweepGenerator] = useState(false);
    72→  const [showBulkTransfer, setShowBulkTransfer] = useState(false);
    73→  const [selectedTransferApp, setSelectedTransferApp] = useState<FantasyApp>("dream11");
    74→  const [showFreeContests, setShowFreeContests] = useState(false);
    75→  const [showOtpLogin, setShowOtpLogin] = useState(false);
    76→  const [linkedApp, setLinkedApp] = useState<FantasyApp | null>(null);
    77→  const [refreshKey, setRefreshKey] = useState(0);
    78→  const [lineupAnnounced, setLineupAnnounced] = useState(false);
    79→  const [playingPlayerIds, setPlayingPlayerIds] = useState<Set<string>>(new Set());
    80→  const [matchSupportsDream11, setMatchSupportsDream11] = useState(false);
    81→
    82→  const matchId = (match.matchId ?? match.id ?? "") as string;
    83→  const team1Name = (typeof match.team1 === "string" ? match.team1 : (match.team1?.shortName || match.team1?.name)) as string ||
    84→    (match as any).left_team_name || "T1";
    85→  const team2Name = (typeof match.team2 === "string" ? match.team2 : (match.team2?.shortName || match.team2?.name)) as string ||
    86→    (match as any).right_team_name || "T2";
    87→  const team1Color = teamColor(team1Name);
    88→  const team2Color = teamColor(team2Name);
    89→
    90→  const loadAll = useCallback(async () => {
    91→    setStep("loading");
    92→    setError(null);
    93→    try {
    94→      // 1. Fetch match detail (real players)
    95→      const matchDetail = await getMatchDetail(matchId).catch(() => null);
    96→
    97→      // Extract players from match detail
    98→      let allPlayers: MatchPlayer[] = [];
    99→      if (matchDetail) {
   100→        const leftPlayers = (matchDetail as any).left_team_players || [];
   101→        const rightPlayers = (matchDetail as any).right_team_players || [];
   102→        allPlayers = [...leftPlayers, ...rightPlayers].filter(Boolean);
   103→
   104→        // Detect lineup status
   105→        const lineupStatus = (matchDetail as any).lineup_status;
   106→        const announced = Number(lineupStatus) === 1;
   107→        setLineupAnnounced(announced);
   108→
   109→        // Build set of confirmed playing player IDs
   110→        const playingSet = buildPlayingPlayerSet(allPlayers, announced);
   111→        setPlayingPlayerIds(playingSet);
   112→
   113→        // Filter players: after lineup, only show confirmed playing XI
   114→        const visiblePlayers = filterPlayingPlayers(allPlayers, announced);
   115→        setPlayers(visiblePlayers);
   116→
   117→        // Detect if match supports Dream11 transfer:
   118→        // Check if fantasy_version includes "dream11" OR any player has dream11 in fantasy_id_list
   119→        const versions = (matchDetail as any).fantasy_version || [];
   120→        const versionSupportsD11 = versions.some((v: any) => v.name === "dream11");
   121→        const playerSupportsD11 = visiblePlayers.some((p: any) => {
   122→          const fil = p.fantasy_id_list || [];
   123→          return fil.some((f: any) => f.name === "dream11");
   124→        });
   125→        setMatchSupportsDream11(versionSupportsD11 || playerSupportsD11);
   126→      } else {
   127→        setPlayers([]);
   128→        setLineupAnnounced(false);
   129→        setPlayingPlayerIds(new Set());
   130→      }
   131→
   132→      // localStorage teams = the user's APNA teams (generated locally)
   133→      const localTeams = getLocalTeams(matchId);
   134→      setExistingTeams(localTeams);
   135→      setStep("ready");
   136→    } catch (err: any) {
   137→      setError(err?.message || "Failed to load match data.");
   138→      setStep("error");
   139→    }
   140→  }, [matchId, token]);
   141→
   142→  useEffect(() => {
   143→    // eslint-disable-next-line react-hooks/set-state-in-effect
   144→    loadAll();
   145→  }, [loadAll, refreshKey]);
   146→
   147→  // Check if user already has a Dream11 token saved
   148→  useEffect(() => {
   149→    const t = getFantasyToken("dream11");
   150→    // eslint-disable-next-line react-hooks/set-state-in-effect
   151→    if (t) setLinkedApp("dream11");
   152→  }, []);
   153→
   154→  async function handleAddTeam(team: Partial<TeamRow>) {
   155→    setStep("saving");
   156→    try {
   157→      const res = await addTeam(token!, { matchId, team, type: "add" });
   158→      const offline = (res as { offline?: boolean }).offline;
   159→      toast.success(offline ? "Team saved locally" : "Team added successfully");
   160→      setShowEditor(false);
   161→      setEditingTeam(null);
   162→      setRefreshKey((k) => k + 1);
   163→    } catch (err: any) {
   164→      toast.error(err?.message || "Failed to add team");
   165→      setStep("ready");
   166→    }
   167→  }
   168→
   169→  async function handleEditTeam(teamId: string, team: Partial<TeamRow>) {
   170→    setStep("saving");
   171→    try {
   172→      const res = await editTeam(token!, { matchId, teamId, team, type: "edit" });
   173→      const offline = (res as { offline?: boolean }).offline;
   174→      toast.success(offline ? "Team updated locally" : "Team updated");
   175→      setShowEditor(false);
   176→      setEditingTeam(null);
   177→      setRefreshKey((k) => k + 1);
   178→    } catch (err: any) {
   179→      toast.error(err?.message || "Failed to update team");
   180→      setStep("ready");
   181→    }
   182→  }
   183→
   184→  async function handleDeleteTeam(teamId: string) {
   185→    setStep("saving");
   186→    try {
   187→      await editTeam(token!, { matchId, teamId, team: {}, type: "delete" });
   188→      toast.success("Team deleted");
   189→      setRefreshKey((k) => k + 1);
   190→    } catch (err: any) {
   191→      toast.error(err?.message || "Failed to delete team");
   192→      setStep("ready");
   193→    }
   194→  }
   195→
   196→  function handleTransfer(team: TeamRow, app: FantasyApp) {
   197→    // Open the APNA Transfer modal for the selected app (same as toolbar buttons)
   198→    if (!license.requireLicense()) return;
   199→    setSelectedTransferApp(app);
   200→    setShowBulkTransfer(true);
   201→  }
   202→
   203→  function handleLinkApp(app: FantasyApp, authToken: string) {
   204→    saveFantasyToken(app, authToken);
   205→    setLinkedApp(app);
   206→    setShowLinkModal(false);
   207→    toast.success(`${app} linked successfully`);
   208→  }
   209→
   210→  // Auto-replace non-playing players in the given teams
   211→  function handleAutoReplaceAll(teamsToFix: TeamRow[]) {
   212→    try {
   213→      const all = JSON.parse(localStorage.getItem("fantasypro-local-teams") || "{}");
   214→      const list = all[matchId] || [];
   215→      let totalReplaced = 0;
   216→
   217→      for (const team of teamsToFix) {
   218→        const idx = list.findIndex(
   219→          (t: TeamRow) => t._id === team._id || t.teamId === team.teamId
   220→        );
   221→        if (idx === -1) continue;
   222→
   223→        const { updatedTeam, replaced } = autoReplaceNonPlaying(
   224→          list[idx],
   225→          players as MatchPlayer[],
   226→          true
   227→        );
   228→        list[idx] = updatedTeam;
   229→        totalReplaced += replaced.length;
   230→      }
   231→
   232→      all[matchId] = list;
   233→      localStorage.setItem("fantasypro-local-teams", JSON.stringify(all));
   234→
   235→      if (totalReplaced > 0) {
   236→        toast.success(`Replaced ${totalReplaced} non-playing player(s) across ${teamsToFix.length} team(s)`);
   237→      } else {
   238→        toast.info("No replacements available — all non-playing players kept (insufficient playing XI)");
   239→      }
   240→      setRefreshKey((k) => k + 1);
   241→    } catch (err: any) {
   242→      toast.error("Failed to auto-replace players");
   243→    }
   244→  }
   245→
   246→  // Called after the auto-generator has finished. The generator already saved
   247→  // teams to localStorage via bulkAddTeams(), so we just need to refresh the
   248→  // view and show a success toast.
   249→  function handleBulkAddTeams(teams: TeamRow[]) {
   250→    toast.success(`Generated ${teams.length} unique teams`);
   251→    setRefreshKey((k) => k + 1);
   252→  }
   253→
   254→  return (
   255→    <div className="space-y-4">
   256→      {/* Status banner */}
   257→      <div className="bg-white dark:bg-card rounded-2xl border border-violet-100 dark:border-violet-900/30 p-4">
   258→        <div className="flex items-center justify-between mb-2">
   259→          <div>
   260→            <h3 className="text-sm font-bold text-foreground flex items-center gap-2">
   261→              My Teams
   262→              {lineupAnnounced ? (
   263→                <span className="inline-flex items-center gap-1 text-[9px] font-bold bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300 px-2 py-0.5 rounded-full">
   264→                  <span className="w-1.5 h-1.5 bg-emerald-500 rounded-full" />
   265→                  PLAYING XI CONFIRMED
   266→                </span>
   267→              ) : (
   268→                <span className="inline-flex items-center gap-1 text-[9px] font-bold bg-amber-100 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300 px-2 py-0.5 rounded-full">
   269→                  <span className="w-1.5 h-1.5 bg-amber-500 rounded-full animate-pulse" />
   270→                  LINEUP PENDING
   271→                </span>
   272→              )}
   273→            </h3>
   274→            <p className="text-[11px] text-muted-foreground">
   275→              Match ID: {matchId || "—"} · {players.length} players available
   276→              {lineupAnnounced && " · only confirmed playing XI shown"}
   277→            </p>
   278→          </div>
   279→          <Button
   280→            variant="outline"
   281→            size="sm"
   282→            onClick={() => setRefreshKey((k) => k + 1)}
   283→            disabled={step === "loading"}
   284→          >
   285→            <RefreshCw className={cn("w-3.5 h-3.5 mr-1.5", step === "loading" && "animate-spin")} />
   286→            Refresh
   287→          </Button>
   288→        </div>
   289→
   290→        {/* Lineup warning — show if teams have non-playing players */}
   291→        {lineupAnnounced && existingTeams.length > 0 && (() => {
   292→          const teamsWithIssues = existingTeams.filter((t) =>
   293→            getNonPlayingPlayersInTeam(t, true, playingPlayerIds).length > 0
   294→          );
   295→          if (teamsWithIssues.length === 0) return null;
   296→          return (
   297→            <div className="mt-3 rounded-xl bg-amber-50 dark:bg-amber-950/30 border border-amber-200 dark:border-amber-800 p-3">
   298→              <p className="text-xs font-bold text-amber-700 dark:text-amber-300 mb-1">
   299→                ⚠️ Lineup Updated — {teamsWithIssues.length} team(s) have non-playing players
   300→              </p>
   301→              <p className="text-[11px] text-muted-foreground mb-2">
   302→                The official Playing XI has been announced. Some of your teams contain players who are not in the final lineup. Auto-replace them now.
   303→              </p>
   304→              <Button
   305→                size="sm"
   306→                className="h-8 text-xs bg-amber-500 hover:bg-amber-600 text-white"
   307→                onClick={() => handleAutoReplaceAll(teamsWithIssues)}
   308→              >
   309→                <RefreshCw className="w-3 h-3 mr-1" />
   310→                Auto-Replace Non-Playing Players
   311→              </Button>
   312→            </div>
   313→          );
   314→        })()}
   315→
   316→        {/* Transfer Flow step indicator */}
   317→        <TransferStepIndicator
   318→          linked={!!linkedApp}
   319→          teamsCount={existingTeams.length}
   320→        />
   321→
   322→        {/* Dream11 link status */}
   323→        <div className="mt-3 flex items-center gap-2 text-[11px]">
   324→          <span
   325→            className={cn(
   326→              "px-2 py-0.5 rounded-full font-semibold",
   327→              linkedApp === "dream11"
   328→                ? "bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300"
   329→                : "bg-violet-50 text-violet-700 dark:bg-violet-900/30 dark:text-violet-300"
   330→            )}
   331→          >
   332→            {linkedApp === "dream11" ? "✓ Dream11 linked" : "Dream11 not linked"}
   333→          </span>
   334→          {!linkedApp && (
   335→            <button
   336→              onClick={() => setShowOtpLogin(true)}
   337→              className="text-violet-600 dark:text-violet-300 font-semibold underline"
   338→            >
   339→              Link via OTP →
   340→            </button>
   341→          )}
   342→        </div>
   343→
   344→        {error && (
   345→          <div className="mt-3 text-xs text-red-600 bg-red-50 dark:bg-red-950/30 p-2.5 rounded-lg flex items-start gap-2">
   346→            <AlertCircle className="w-4 h-4 flex-shrink-0 mt-0.5" />
   347→            <span>{error}</span>
   348→          </div>
   349→        )}
   350→
   351→        {step === "ready" && existingTeams.length === 0 && !showEditor && (
   352→          <div className="mt-3 space-y-3">
   353→            <div className="text-center py-2">
   354→              <p className="text-sm text-muted-foreground mb-3">
   355→                No teams yet. Auto-generate up to 500 teams or build one manually from {players.length} real players.
   356→              </p>
   357→              {/* License lock banner */}
   358→              {!license.isLicensed && (
   359→                <div className="mb-3 rounded-xl bg-amber-50 dark:bg-amber-950/30 border border-amber-300 dark:border-amber-800 p-3 text-center">
   360→                  <div className="flex items-center justify-center gap-1.5 mb-1">
   361→                    <Lock className="w-4 h-4 text-amber-600" />
   362→                    <p className="text-xs font-bold text-amber-700 dark:text-amber-300">
   363→                      License Required to Generate Teams
   364→                    </p>
   365→                  </div>
   366→                  <p className="text-[10px] text-muted-foreground">
   367→                    Activate your CRICB license key to unlock team generation, auto pick, GL generator & more.
   368→                  </p>
   369→                </div>
   370→              )}
   371→              <div className="flex flex-col sm:flex-row gap-2 justify-center">
   372→                <Button
   373→                  onClick={() => license.requireLicense() && setShowSweepGenerator(true)}
   374→                  className={cn(license.isLicensed ? "brand-gradient text-white" : "bg-slate-300 text-slate-500 dark:bg-slate-700 dark:text-slate-400")}
   375→                  disabled={players.length < 11}
   376→                >
   377→                  {license.isLicensed ? <Zap className="w-4 h-4 mr-1.5" /> : <Lock className="w-4 h-4 mr-1.5" />}
   378→                  {license.isLicensed ? "Auto Generate Teams" : "Auto Generate (Locked)"}
   379→                </Button>
   380→                <Button
   381→                  variant="outline"
   382→                  onClick={() => license.requireLicense() && (() => {
   383→                    setEditingTeam(null);
   384→                    setShowEditor(true);
   385→                  })()}
   386→                  disabled={players.length === 0}
   387→                  className={cn(!license.isLicensed && "border-amber-300 text-amber-600 dark:border-amber-800 dark:text-amber-400")}
   388→                >
   389→                  {license.isLicensed ? <Plus className="w-4 h-4 mr-1.5" /> : <Lock className="w-4 h-4 mr-1.5" />}
   390→                  {license.isLicensed ? "Build Manually" : "Build (Locked)"}
   391→                </Button>
   392→              </div>
   393→            </div>
   394→          </div>
   395→        )}
   396→      </div>
   397→
   398→      {/* Existing teams list */}
   399→      {step === "ready" && existingTeams.length > 0 && !showEditor && (
   400→        <div className="space-y-3">
   401→          <div className="flex items-center justify-between flex-wrap gap-2">
   402→            <p className="text-xs font-semibold text-muted-foreground">
   403→              {existingTeams.length} team{existingTeams.length !== 1 ? "s" : ""} saved
   404→            </p>
   405→            <div className="flex gap-1.5 flex-wrap">
   406→              {/* Transfer to Dream11 — always show */}
   407→              <Button
   408→                size="sm"
   409→                onClick={() => {
   410→                  if (!license.requireLicense()) return;
   411→                  setSelectedTransferApp("dream11");
   412→                  setShowBulkTransfer(true);
   413→                }}
   414→                className={cn(
   415→                  license.isLicensed ? "bg-red-600 hover:bg-red-700 text-white" : "bg-slate-300 text-slate-500 dark:bg-slate-700",
   416→                  !license.isLicensed && "cursor-pointer"
   417→                )}
   418→                disabled={existingTeams.length === 0}
   419→              >
   420→                {license.isLicensed ? <Send className="w-3.5 h-3.5 mr-1" /> : <Lock className="w-3.5 h-3.5 mr-1" />}
   421→                {license.isLicensed ? "🏏 Dream11" : "Dream11 🔒"}
   422→              </Button>
   423→              {/* Transfer to My11Circle — always show */}
   424→              <Button
   425→                size="sm"
   426→                onClick={() => {
   427→                  if (!license.requireLicense()) return;
   428→                  setSelectedTransferApp("my11circle");
   429→                  setShowBulkTransfer(true);
   430→                }}
   431→                className={cn(
   432→                  license.isLicensed ? "bg-sky-600 hover:bg-sky-700 text-white" : "bg-slate-300 text-slate-500 dark:bg-slate-700",
   433→                  !license.isLicensed && "cursor-pointer"
   434→                )}
   435→                disabled={existingTeams.length === 0}
   436→              >
   437→                {license.isLicensed ? <Send className="w-3.5 h-3.5 mr-1" /> : <Lock className="w-3.5 h-3.5 mr-1" />}
   438→                {license.isLicensed ? "⭕ My11Circle" : "My11 🔒"}
   439→              </Button>
   440→              <Button
   441→                size="sm"
   442→                onClick={() => license.requireLicense() && setShowFreeContests(true)}
   443→                className={cn(
   444→                  license.isLicensed ? "bg-amber-500 hover:bg-amber-600 text-white" : "bg-slate-300 text-slate-500 dark:bg-slate-700"
   445→                )}
   446→                disabled={existingTeams.length === 0}
   447→              >
   448→                {license.isLicensed ? <Gift className="w-3.5 h-3.5 mr-1" /> : <Lock className="w-3.5 h-3.5 mr-1" />}
   449→                {license.isLicensed ? "Free Contests" : "Contests 🔒"}
   450→              </Button>
   451→              <Button
   452→                size="sm"
   453→                variant="outline"
   454→                onClick={() => license.requireLicense() && setShowSweepGenerator(true)}
   455→                disabled={players.length < 11}
   456→                className={cn(
   457→                  "border-violet-200 text-violet-700 dark:text-violet-300",
   458→                  !license.isLicensed && "border-amber-300 text-amber-600 dark:border-amber-800 dark:text-amber-400"
   459→                )}
   460→              >
   461→                {license.isLicensed ? <Zap className="w-3.5 h-3.5 mr-1" /> : <Lock className="w-3.5 h-3.5 mr-1" />}
   462→                {license.isLicensed ? "Auto Generate" : "Generate 🔒"}
   463→              </Button>
   464→              <Button
   465→                size="sm"
   466→                onClick={() => license.requireLicense() && (() => {
   467→                  setEditingTeam(null);
   468→                  setShowEditor(true);
   469→                })()}
   470→                className={cn(
   471→                  license.isLicensed ? "brand-gradient text-white" : "bg-slate-300 text-slate-500 dark:bg-slate-700"
   472→                )}
   473→                disabled={players.length === 0}
   474→              >
   475→                {license.isLicensed ? <Plus className="w-4 h-4 mr-1" /> : <Lock className="w-4 h-4 mr-1" />}
   476→                {license.isLicensed ? "Add Team" : "Add 🔒"}
   477→              </Button>
   478→            </div>
   479→          </div>
   480→
   481→          {existingTeams.map((t, i) => (
   482→            <motion.div
   483→              key={(t._id ?? t.teamId ?? i) as string}
   484→              initial={{ opacity: 0, y: 10 }}
   485→              animate={{ opacity: 1, y: 0 }}
   486→              transition={{ delay: i * 0.05 }}
   487→              className="space-y-2"
   488→            >
   489→              <div className="flex items-center justify-between px-1">
   490→                <span className="text-xs font-bold text-foreground flex items-center gap-1.5">
   491→                  Team #{t.teamNo ?? i + 1}
   492→                  {t.transferred && (
   493→                    <span className="text-[9px] bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300 px-1.5 py-0.5 rounded-full">
   494→                      TRANSFERRED
   495→                    </span>

   785→function TeamEditor({
   786→  team,
   787→  match,
   788→  players,
   789→  team1Name,
   790→  team2Name,
   791→  team1Color,
   792→  team2Color,
   793→  lineupAnnounced,
   794→  onClose,
   795→  onSave,
   796→  saving,
   797→}: {
   798→  team: TeamRow | null;
   799→  match: MatchInfo;
   800→  players: MatchPlayer[];
   801→  team1Name: string;
   802→  team2Name: string;
   803→  team1Color: string;
   804→  team2Color: string;
   805→  lineupAnnounced: boolean;
   806→  onClose: () => void;
   807→  onSave: (team: Partial<TeamRow>) => void;
   808→  saving: boolean;
   809→}) {
   810→  // Initialize selected player IDs from the team being edited
   811→  const [selectedIds, setSelectedIds] = useState<Set<string>>(
   812→    new Set(
   813→      (team?.players || []).map((p) => String(p.player_fixed_id ?? p.pl_id ?? p.pid ?? p.name))
   814→    )
   815→  );
   816→  const [captain, setCaptain] = useState(
   817→    team ? String(team.captain ?? team.c ?? "") : ""
   818→  );
   819→  const [viceCaptain, setViceCaptain] = useState(
   820→    team ? String(team.viceCaptain ?? team.vc ?? team.vicecaptain ?? "") : ""
   821→  );
   822→  const [search, setSearch] = useState("");
   823→  const [filterTeam, setFilterTeam] = useState<string>("all");
   824→  const [filterRole, setFilterRole] = useState<string>("all");
   825→
   826→  const matchId = (match.matchId ?? match.id ?? "") as string;
   827→
   828→  const filteredPlayers = players.filter((p) => {
   829→    const name = String(p.name || "");
   830→    if (search && !name.toLowerCase().includes(search.toLowerCase())) return false;
   831→    if (filterTeam !== "all" && p.team_name !== filterTeam) return false;
   832→    if (filterRole !== "all") {
   833→      const r = roleFromPlayerType(p.player_type);
   834→      if (r !== filterRole) return false;
   835→    }
   836→    return true;
   837→  });
   838→
   839→  function togglePlayer(p: MatchPlayer) {
   840→    const id = String(p.player_fixed_id ?? p.pl_id ?? p.pid ?? p.name);
   841→    const next = new Set(selectedIds);
   842→    if (next.has(id)) {
   843→      next.delete(id);
   844→      // Reset C/VC if removed
   845→      if (captain === id) setCaptain("");
   846→      if (viceCaptain === id) setViceCaptain("");
   847→    } else {
   848→      if (next.size >= 11) {
   849→        toast.error("Maximum 11 players selected");
   850→        return;
   851→      }
   852→      next.add(id);
   853→      // Auto-assign C/VC to first two
   854→      if (!captain) setCaptain(id);
   855→      else if (!viceCaptain && id !== captain) setViceCaptain(id);
   856→    }
   857→    setSelectedIds(next);
   858→  }
   859→
   860→  const selectedPlayers = players.filter((p) =>
   861→    selectedIds.has(String(p.player_fixed_id ?? p.pl_id ?? p.pid ?? p.name))
   862→  );
   863→
   864→  const t1Count = selectedPlayers.filter((p) => p.team_name === team1Name).length;
   865→  const t2Count = selectedPlayers.filter((p) => p.team_name === team2Name).length;
   866→  const totalCredits = selectedPlayers.reduce((sum, p) => sum + (Number(p.credits) || 0), 0);
   867→
   868→  return (
   869→    <motion.div
   870→      initial={{ opacity: 0 }}
   871→      animate={{ opacity: 1 }}
   872→      exit={{ opacity: 0 }}
   873→      className="fixed inset-0 z-50 bg-black/50 backdrop-blur-sm flex items-end sm:items-center justify-center p-0 sm:p-4"
   874→      onClick={onClose}
   875→    >
   876→      <motion.div
   877→        initial={{ y: 40, opacity: 0 }}
   878→        animate={{ y: 0, opacity: 1 }}
   879→        exit={{ y: 40, opacity: 0 }}
   880→        className="bg-white dark:bg-background w-full sm:max-w-lg rounded-t-3xl sm:rounded-3xl shadow-2xl max-h-[95vh] overflow-y-auto"
   881→        onClick={(e) => e.stopPropagation()}
   882→      >
   883→        {/* Header */}
   884→        <div className="sticky top-0 brand-gradient text-white p-4 flex items-center justify-between z-10">
   885→          <h3 className="font-bold flex items-center gap-2">
   886→            {team ? "Edit Team" : "Build New Team"}
   887→            <span className="text-[10px] bg-white/20 px-2 py-0.5 rounded-full">
   888→              {selectedIds.size}/11
   889→            </span>
   890→          </h3>
   891→          <button onClick={onClose} className="p-1 rounded-full hover:bg-white/20">
   892→            <X className="w-5 h-5" />
   893→          </button>
   894→        </div>
   895→
   896→        <div className="p-4 space-y-4">
   897→          {/* Lineup status notice */}
   898→          {lineupAnnounced ? (
   899→            <div className="rounded-xl bg-emerald-50 dark:bg-emerald-950/30 border border-emerald-200 dark:border-emerald-800 p-3 flex items-start gap-2">
   900→              <CheckCircle2 className="w-4 h-4 text-emerald-600 dark:text-emerald-400 mt-0.5 flex-shrink-0" />
   901→              <div>
   902→                <p className="text-xs font-bold text-emerald-700 dark:text-emerald-300">
   903→                  Playing XI Confirmed
   904→                </p>
   905→                <p className="text-[10px] text-muted-foreground mt-0.5">
   906→                  Only confirmed playing XI players are shown. Bench, retired, and unavailable players are hidden.
   907→                </p>
   908→              </div>
   909→            </div>
   910→          ) : (
   911→            <div className="rounded-xl bg-amber-50 dark:bg-amber-950/30 border border-amber-200 dark:border-amber-800 p-3 flex items-start gap-2">
   912→              <AlertCircle className="w-4 h-4 text-amber-600 dark:text-amber-400 mt-0.5 flex-shrink-0" />
   913→              <div>
   914→                <p className="text-xs font-bold text-amber-700 dark:text-amber-300">

   924→          {/* Squad summary */}
   925→          <div className="grid grid-cols-4 gap-2 text-center">
   926→            <div className="rounded-lg bg-violet-50 dark:bg-violet-900/20 p-2">

   929→            </div>
