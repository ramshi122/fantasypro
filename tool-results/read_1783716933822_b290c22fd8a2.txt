    21→    21→ * (NOT a collection download) — handled in `verifyLicense` via

   218→   218→export function getExpiryForType(type: KeyType): number | null {
   219→   219→  const now = Date.now();
   220→   220→  const day = 86400000;
   221→   221→  const hour = 3600000;
   222→   222→  switch (type) {

   224→   224→    case "match_basic": return now + 6 * hour; // 1 match = 6 hours
   225→   225→    case "match_full": return now + 1 * day;   // 24 hour active
   226→   226→    case "weekly": return now + 7 * day;       // 7 days
   227→   227→    case "monthly": return now + 30 * day;     // 30 days (Pro ₹299)
   228→   228→    case "elite": return now + 90 * day;       // 90 days (Elite ₹999)
   229→   229→    case "lifetime": return null;              // forever
   230→   230→  }

   239→   239→function lsGet<T>(key: string, fallback: T): T {
   240→   240→  if (typeof window === "undefined") return fallback;
   241→   241→  try {
   242→   242→    const raw = localStorage.getItem(key);
   243→   243→    if (!raw) return fallback;
   244→   244→    return JSON.parse(raw) as T;
   245→   245→  } catch {
   246→   246→    return fallback;
   247→   247→  }
   248→   248→}

   250→   250→function lsSet<T>(key: string, value: T): void {
   251→   251→  if (typeof window === "undefined") return;
   252→   252→  try {
   253→   253→    localStorage.setItem(key, JSON.stringify(value));
   254→   254→  } catch {
   255→   255→    /* ignore quota errors */
   256→   256→  }
   257→   257→}

   368→   368→  const persistedUsers = lsGet<User[] | null>(LS_USERS, null);
