   210→function AdminLoginModal({ open, onClose }: { open: boolean; onClose: () => void }) {
   211→  const admin = useAdmin();
   212→  const [password, setPassword] = useState("");
   213→  const [show, setShow] = useState(false);
   214→  const [loading, setLoading] = useState(false);
   215→
   216→  if (!open) return null;
   217→
   218→  function handleLogin() {

   220→      toast.error("Enter password");

   225→      const ok = admin.login("", password);

   309→/* ---------- Dashboard Tab ---------- */

   311→function DashboardTab() {
   312→  // Select raw data and compute stats in useMemo to avoid infinite loop
   313→  const keys = useAdmin((s) => s.keys);
   314→  const devices = useAdmin((s) => s.devices);
   315→  const users = useAdmin((s) => s.users);
   316→  const logs = useAdmin((s) => s.logs);
   317→  const isSyncing = useAdmin((s) => s.isSyncing);
   318→  const lastSyncedAt = useAdmin((s) => s.lastSyncedAt);
   319→  const syncFromFirebase = useAdmin((s) => s.syncFromFirebase);
   320→  const subscribeToFirebase = useAdmin((s) => s.subscribeToFirebase);
   321→  const isSubscribed = useAdmin((s) => s.isSubscribed);
   322→  const [syncing, setSyncing] = useState(false);
   323→
   324→  // Sync keys from Firebase — ensures all admin panels show SAME keys.
   325→  // Delegates to the store's syncFromFirebase which fetches ALL admin data
   326→  // (keys, devices, users, logs, settings) from the single source of truth.
   327→  async function handleSync() {
   328→    setSyncing(true);
   329→    try {
   330→      const ok = await syncFromFirebase();
   331→      // Re-subscribe (idempotent — no-op if already listening)
   332→      subscribeToFirebase();
   333→      if (ok) {
   334→        const total = useAdmin.getState().keys.length;
   335→        toast.success(`✓ Synced from Firebase — ${total} keys total`);
   336→      } else {
   337→        toast.error("Sync failed — Firebase may be offline");
   338→      }
   339→    } catch (err: any) {
   340→      toast.error(err?.message || "Sync failed");
   341→    } finally {
   342→      setSyncing(false);
   343→    }
   344→  }
   345→
   346→  const lastSyncedLabel = (() => {
   347→    if (!lastSyncedAt) return "Never";
   348→    const diff = Date.now() - lastSyncedAt;
   349→    if (diff < 60_000) return "just now";
   350→    if (diff < 3_600_000) return `${Math.floor(diff / 60_000)}m ago`;
   351→    return new Date(lastSyncedAt).toLocaleTimeString();
   352→  })();
   353→
   354→  const stats = (() => {
   355→    const now = Date.now();
   356→    const todayStart = new Date().setHours(0, 0, 0, 0);
   357→    return {
   358→      totalKeys: keys.length,
   359→      activeKeys: keys.filter((k) => k.status === "active" && (!k.expiresAt || k.expiresAt > now)).length,
   360→      usedKeys: keys.filter((k) => k.status === "used").length,
   361→      expiredKeys: keys.filter((k) => k.status === "expired" || (k.expiresAt && k.expiresAt < now)).length,
   362→      activeDevices: devices.filter((d) => d.status === "allowed").length,
   363→      todayVerifications: logs.filter((l) => l.type === "key_verification" && l.timestamp >= todayStart).length,
   364→      totalUsers: users.length,
   365→      teamsGeneratedToday: logs.filter((l) => l.type === "team_generation" && l.timestamp >= todayStart).length,
   366→    };
   367→  })();
   368→
   369→  const cards = [
   370→    { label: "Total Keys", value: stats.totalKeys, icon: Key, color: "from-violet-500 to-purple-600" },
   371→    { label: "Active Keys", value: stats.activeKeys, icon: ShieldCheck, color: "from-emerald-500 to-green-600" },
   372→    { label: "Used Keys", value: stats.usedKeys, icon: Check, color: "from-sky-500 to-blue-600" },
   373→    { label: "Expired Keys", value: stats.expiredKeys, icon: AlertTriangle, color: "from-amber-500 to-orange-600" },
   374→    { label: "Active Devices", value: stats.activeDevices, icon: Smartphone, color: "from-pink-500 to-rose-600" },
   375→    { label: "Today Verifications", value: stats.todayVerifications, icon: Activity, color: "from-cyan-500 to-teal-600" },
   376→    { label: "Total Users", value: stats.totalUsers, icon: Users, color: "from-indigo-500 to-violet-600" },
   377→    { label: "Teams Today", value: stats.teamsGeneratedToday, icon: Zap, color: "from-fuchsia-500 to-pink-600" },
   378→  ];
   379→  return (
   380→    <div className="space-y-4">
   381→      <div className="flex items-center justify-between">
   382→        <h3 className="text-sm font-bold text-white">Dashboard Overview</h3>
   383→        <Button
   384→          size="sm"
   385→          className="bg-emerald-600 hover:bg-emerald-700 text-white h-8 text-xs"
   386→          onClick={handleSync}
   387→          disabled={syncing || isSyncing}
   388→        >
   389→          {syncing || isSyncing ? <Loader2 className="w-3 h-3 mr-1 animate-spin" /> : <RefreshCw className="w-3 h-3 mr-1" />}
   390→          Sync from Firebase
   391→        </Button>
   392→      </div>
   393→      <div className="bg-emerald-900/20 border border-emerald-800/50 rounded-lg p-2 text-[10px] text-emerald-300 flex items-center justify-between flex-wrap gap-1">
   394→        <span className="flex items-center gap-1.5">
   395→          <span className={cn("w-1.5 h-1.5 rounded-full", isSubscribed ? "bg-emerald-400 animate-pulse" : "bg-emerald-600")} />
   396→          ✓ Synced from Firebase — keys, plans, and data are the SAME across all devices/APKs
   397→        </span>
   398→        <span className="text-emerald-400/80">
   399→          {isSubscribed ? "● Live" : "○ Offline"} · Last sync: {lastSyncedLabel}
   400→        </span>
   401→      </div>
   402→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   403→        {cards.map((c) => {
   404→          const Icon = c.icon;
   405→          return (
   406→            <div key={c.label} className="bg-slate-800 rounded-xl p-3 border border-slate-700">
   407→              <div className={cn("w-8 h-8 rounded-lg bg-gradient-to-br flex items-center justify-center mb-2", c.color)}>
   408→                <Icon className="w-4 h-4 text-white" />
   409→              </div>
   410→              <p className="text-2xl font-black text-white">{c.value}</p>
   411→              <p className="text-[10px] text-slate-400 font-semibold uppercase tracking-wide">{c.label}</p>
   412→            </div>
   413→          );
   414→        })}
   415→      </div>
   416→    </div>
   417→  );
   418→}
