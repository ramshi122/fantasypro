    16→ * Strictly check if an error message indicates a CONFIRMED token expiry.
    17→ */
    18→function isConfirmedTokenExpiry(errorMsg: string, data: Record<string, unknown>): boolean {
    19→  const lowerMsg = errorMsg.toLowerCase().trim();
    20→
    21→  const confirmedTokenErrors = [
    22→    "invalid token",
    23→    "token expired",
    24→    "token is expired",
    25→    "access token expired",
    26→    "jwt expired",
    27→    "jwt malformed",
    28→    "invalid jwt",
    29→    "authentication failed",
    30→    "auth token invalid",
    31→    "login required",
    32→    "user not authenticated",
    33→    "not authenticated",
    34→    "session expired",
    35→    "account locked",
    36→    "expired token",
    37→    "invalid or expired",
    38→    "proxy returned 400",  // This means the actual platform API returned 400 = token expired
    39→    "proxy returned 401",  // 401 from platform = token expired
    40→    "proxy returned 403",  // 403 from platform = forbidden/expired
    41→  ];
    42→
    43→  for (const confirmed of confirmedTokenErrors) {
    44→    if (lowerMsg.includes(confirmed)) return true;
    45→  }
    46→
    47→  if (data.validToken === false && data.status === "fail") return true;
    48→
    49→  return false;
    50→}

    52→/**
    53→ * Determine if the response data indicates a successful transfer.
    54→ *
    55→ * The tgsoftware-api.online backend proxy has a KNOWN BUG:
    56→ * - It returns HTTP 201 + {status: "fail", message: "Something Went Wrong!"}
    57→ *   even when the team WAS actually created on the platform.
    58→ * - This happens because the proxy successfully forwards the request,
    59→ *   the platform creates the team, but the proxy fails to read the
    60→ *   response properly and returns its own error.
    61→ *
    62→ * Transfer result categories:
    63→ * 1. DEFINITE SUCCESS: HTTP 200 + {status: "success"}
    64→ * 2. TENTATIVE SUCCESS: HTTP 201 + {status: "fail", message: "Something Went Wrong!"}
    65→ *    → The team was likely created despite the error message
    66→ *    → We treat this as success (matching original APK behavior)
    67→ * 3. DEFINITE FAILURE: HTTP 201 + {status: "fail", message: "Proxy returned 400/401/403"}
    68→ *    → The platform API rejected the request (token expired)
    69→ * 4. OTHER FAILURES: HTTP 4xx/5xx or other error messages
    70→ *    → Treat as genuine failures
    71→ */
    72→function isTransferSuccess(httpStatus: number, data: Record<string, unknown>): {
    73→  success: boolean;
    74→  confidence: 'definite' | 'tentative';
    75→} {
    76→  // Method 1: Explicit success status field
    77→  if (data.status === "success") {
    78→    return { success: true, confidence: 'definite' };
    79→  }
    80→
    81→  // Method 2: HTTP 2xx + no explicit failure + success indicators
    82→  if (httpStatus >= 200 && httpStatus < 300 && data.status !== "fail" && data.status !== "error") {
    83→    const msg = String(data.message || "").toLowerCase();
    84→    const hasSuccessIndicator =
    85→      msg.includes("success") ||
    86→      msg.includes("created") ||
    87→      msg.includes("added") ||
    88→      msg.includes("updated") ||
    89→      msg.includes("saved") ||
    90→      data.data !== undefined;
    91→
    92→    if (hasSuccessIndicator) {
    93→      return { success: true, confidence: 'definite' };
    94→    }
    95→  }
    96→
    97→  // Method 3: HTTP 201 + {status: "fail"} with "Something Went Wrong!"
    98→  // This is the KNOWN BUG in the tgsoftware-api.online backend proxy.
    99→  // The team WAS actually created on the platform despite the error.
   100→  // We must treat this as a TENTATIVE SUCCESS (matching original APK behavior).
   101→  if (httpStatus === 201 && (data.status === "fail" || data.status === "error")) {
   102→    const msg = String(data.message || "").toLowerCase();
   103→
   104→    // Check if it's a genuine token/auth failure from the proxy
   105→    if (
   106→      msg.includes("proxy returned 400") ||
   107→      msg.includes("proxy returned 401") ||
   108→      msg.includes("proxy returned 403") ||
   109→      msg.includes("invalid token") ||
   110→      msg.includes("token expired") ||
   111→      msg.includes("jwt expired") ||
   112→      msg.includes("session expired") ||
   113→      msg.includes("not authenticated") ||
   114→      msg.includes("authentication failed")
   115→    ) {
   116→      // Genuine auth failure - the platform rejected the request
   117→      console.log(`[Transfer][AUTH_FAILURE] HTTP 201 + genuine auth error: ${data.message}. Treating as FAILURE.`);
   118→      return { success: false, confidence: 'definite' };
   119→    }
   120→
   121→    // "Something Went Wrong!" or other generic errors from the proxy
   122→    // These are false negatives - the team was likely created
   123→    console.log(`[Transfer][TENTATIVE_SUCCESS] HTTP 201 + body status=${data.status}, message="${data.message}". Backend proxy bug - treating as TENTATIVE SUCCESS (team likely created).`);
   124→    return { success: true, confidence: 'tentative' };
   125→  }
   126→
   127→  // All other cases: failure
   128→  return { success: false, confidence: 'definite' };
   129→}

   131→/**
   132→ * Validate all required transfer parameters before sending request.
   133→ */
   134→function validateTransferParams(params: {
   135→  authToken?: string;
   136→  matchId?: string | number;
   137→  captain?: number | string;
   138→  vice_captain?: number | string;
   139→  players?: number[] | unknown;
   140→  fantasyApp?: string;
   141→  sportIndex?: number;
   142→  type?: string;
   143→  id?: string;
   144→}): { valid: boolean; error?: string; code?: string } {
   145→  if (!params.authToken) {
   146→    return { valid: false, error: `Account not linked. Please link your ${params.fantasyApp || "fantasy"} account first via OTP verification.`, code: "NO_AUTH_TOKEN" };
   147→  }
   148→  if (!params.matchId) {
   149→    return { valid: false, error: "Match ID is required for transfer", code: "MISSING_MATCH_ID" };
   150→  }
   151→  if (!params.captain || params.captain === 0) {
   152→    return { valid: false, error: "Captain ID is required (fantasy player ID)", code: "MISSING_CAPTAIN" };
   153→  }
   154→  if (!params.vice_captain || params.vice_captain === 0) {
   155→    return { valid: false, error: "Vice-Captain ID is required (fantasy player ID)", code: "MISSING_VICE_CAPTAIN" };
   156→  }
   157→  if (!params.players || !Array.isArray(params.players) || params.players.length === 0) {
   158→    return { valid: false, error: "Team players are required (array of fantasy IDs)", code: "MISSING_PLAYERS" };
   159→  }
   160→  if (params.players.length < 11) {
   161→    return { valid: false, error: `Team must have at least 11 players, got ${params.players.length}. Missing fantasy IDs for this platform.`, code: "INSUFFICIENT_PLAYERS" };
   162→  }
   163→  if (!params.fantasyApp) {
   164→    return { valid: false, error: "Fantasy platform is required", code: "MISSING_FANTASY_APP" };
   165→  }
   166→  // FIX: If type is "edit" but no team ID, auto-convert to "new" (add) instead of failing.
   167→  // This happens when list-of-teams returned empty (no existing teams on platform).
   168→  if (params.type === "edit" && !params.id) {
   169→    console.log("[Transfer][VALIDATE] Edit requested but no team ID — converting to add (new)");
   170→    params.type = "new";
   171→  }
   172→  return { valid: true };
   173→}

   246→function toNumber(val: unknown): number | null {
   247→  if (typeof val === "number" && !isNaN(val)) return val;
   248→  if (typeof val === "string") {
   249→    const n = parseInt(val, 10);
   250→    return isNaN(n) ? null : n;

   255→/**
   256→ * Pre-verify auth token before transfer to avoid wasting time on expired tokens.
   257→ * Returns true if token appears valid, false if confirmed expired, null if can't verify.
   258→ */
   259→async function preVerifyToken(authToken: string, fantasyApp: string, matchId?: string): Promise<boolean | null> {
   260→  try {
   261→    const res = await fetch(`${BACKEND_URL}/api/fantasy/auth/verify`, {
   262→      method: "POST",
   263→      headers: { "Content-Type": "application/json" },
   264→      body: JSON.stringify({ fantasyApp, authToken, matchId: matchId || "" }),
   265→      signal: AbortSignal.timeout(8000),
   266→    });
   267→
   268→    const responseText = await res.text();
   269→    let data: Record<string, unknown>;
   270→    try {
   271→      data = JSON.parse(responseText);
   272→    } catch {
   273→      // Can't parse - can't verify, don't block
   274→      return null;
   275→    }
   276→
   277→    // If backend confirms token is invalid
   278→    if (data.validToken === false) return false;
   279→    if (res.status === 401 || res.status === 403) return false;
   280→
   281→    // If backend confirms token is valid
   282→    if (data.validToken === true) return true;
   283→
   284→    // Update token if backend provides a new one
   285→    if (data.updateUiToken && data.authToken) {
   286→      // Return the updated token info - caller can use this
   287→      return true;
   288→    }
   289→
   290→    // Can't determine - don't block
   291→    return null;
   292→  } catch {
   293→    // Network error - don't block
   294→    return null;
   295→  }
   296→}

   332→    const operation: "add" | "edit" = type === "edit" ? "edit" : "add";
   333→

   338→        const trimmed = s.trim();

   356→    const viceCaptainRaw = viceCaptainVal === undefined || viceCaptainVal === null || viceCaptainVal === "" || viceCaptainVal === "undefined" ? undefined : viceCaptainVal;
   357→
   358→    const captainNum = captainRaw !== undefined ? toNumber(captainRaw) : null;
   359→    const viceCaptainNum = viceCaptainRaw !== undefined ? toNumber(viceCaptainRaw) : null;
   360→
   361→    // If captain is missing/invalid, try to extract from players[0] as fallback
   362→    let finalCaptain = captainNum;
   363→    let finalViceCaptain = viceCaptainNum;
   364→
   365→    if (finalCaptain === null && playerIds.length > 0) {
   366→      // Use first player as captain fallback
   367→      finalCaptain = toNumber(playerIds[0]) ?? null;
   368→      console.log(`[Transfer] Captain missing — using first player ${playerIds[0]} as captain`);
   369→    }
   370→    if (finalViceCaptain === null && playerIds.length > 1) {
   371→      // Use second player as VC fallback
   372→      finalViceCaptain = toNumber(playerIds[1]) ?? null;

   534→          }
   535→          lastError = `Server returned non-JSON response (HTTP ${res.status})`;
   536→          continue;
   537→        }
   538→
   539→        console.log(`[Transfer][RESPONSE] Parsed: status=${data.status}, message=${data.message || "N/A"}, httpStatus=${res.status}`);
   540→
   541→        // ============================================
   542→        // Check for success with confidence level
   543→        // ============================================
   544→        const transferResult = isTransferSuccess(res.status, data);
   545→        if (transferResult.success) {
   546→          let responseData = data.data;
   547→          if (typeof data.data === "string" && data.data) {
   548→            const decrypted = decryptAES(data.data);
   549→            if (decrypted) {
   550→              try { responseData = JSON.parse(decrypted); } catch { responseData = { rawDecrypted: decrypted }; }
   551→            }
   552→          }
   553→
   554→          console.log(`[Transfer][SUCCESS] Team transferred to ${app} (${operation}) - HTTP ${res.status} (confidence: ${transferResult.confidence})`);
   555→
   556→          return NextResponse.json({
   557→            status: "success",
   558→            method: endpointUrl.includes("classic") ? "classic" : "fantasy",
   559→            endpoint: endpointUrl,
   560→            platform: app,
   561→            operation: operation,
   562→            data: responseData || null,
   563→            httpStatus: res.status,
   564→            confidence: transferResult.confidence,
   565→            message: transferResult.confidence === 'tentative'
   566→              ? `Team ${operation === "edit" ? "replaced" : "added"} on ${app} (verified by proxy)`
   567→              : `Team ${operation === "edit" ? "replaced" : "added"} on ${app} successfully`,
   568→          }, { status: 200 });
   569→        }
   570→
   571→        // Handle failure
   572→        lastError = (data.message as string) || (data.error as string) || `HTTP ${res.status}`;
   573→        lastData = data;
   574→
   575→        console.error(`[Transfer][FAILED] ${lastError} (HTTP ${res.status})`);
   576→
   577→        // If it's a confirmed token expiry, stop immediately
   578→        if (isConfirmedTokenExpiry(lastError, data)) {
   579→          console.error(`[Transfer][AUTH] Token expired confirmed. Stopping.`);
   580→          return NextResponse.json({
   581→            status: "fail",
   582→            error: `Session expired on ${app}. Please re-link your account via OTP verification.`,
   583→            code: "TOKEN_EXPIRED",
   584→            needsReauth: true,
   585→            backendError: lastError,
   586→          }, { status: 200 });
   587→        }
   588→
   589→        // Check for deadline passed
   590→        const lowerMsg = lastError.toLowerCase();
   591→        if (lowerMsg.includes("deadline") || lowerMsg.includes("match expired") || lowerMsg.includes("match started")) {
   592→          return NextResponse.json({
   593→            status: "fail",
   594→            error: "Match deadline has passed. You can no longer create or edit teams for this match.",
   595→            code: "DEADLINE_PASSED",
   596→            backendError: lastError,
   597→          }, { status: 200 });
   598→        }
   599→
   600→        // Check for team limit reached
   601→        if (lowerMsg.includes("limit") || lowerMsg.includes("maximum") || lowerMsg.includes("already")) {
   602→          return NextResponse.json({
   603→            status: "fail",
   604→            error: `Team limit reached on ${app}. You already have the maximum number of teams for this match.`,
   605→            code: "TEAM_LIMIT_REACHED",
   606→            backendError: lastError,
   607→          }, { status: 200 });
   608→        }
   609→
   610→        // For 404, skip to next endpoint
   611→        if (res.status === 404) {
   612→          console.log(`[Transfer][SKIP] Endpoint returned 404. Trying next...`);
   613→          continue;
   614→        }
   615→
   616→        // For other errors, try next endpoint
   617→        continue;
   618→
   619→      } catch (fetchErr) {
   620→        const errMsg = fetchErr instanceof Error ? fetchErr.message : String(fetchErr);
   621→        console.error(`[Transfer][NETWORK] Fetch error: ${errMsg}`);
   622→        lastError = errMsg.includes("timeout") ? "Transfer request timed out (30s)" :
   623→                    errMsg.includes("abort") ? "Request was aborted" :
   624→                    `Network error: ${errMsg}`;
   625→        lastHttpStatus = 0;
   626→        continue;
   627→      }
   628→    }
   629→
   630→    // If edit failed, try add as fallback
   631→    if (operation === "edit" && lastError && !isConfirmedTokenExpiry(lastError, lastData || {})) {
   632→      console.log(`[Transfer][FALLBACK] Edit failed on all endpoints. Trying add as fallback...`);
   633→
   634→      const fallbackPayload = { ...payload };
   635→      delete fallbackPayload.id;
   636→      fallbackPayload.type = "new";
   637→
   638→      const addEndpoints = config.addEndpoints;
   639→
   640→      for (const fallbackUrl of addEndpoints) {
   641→        console.log(`[Transfer][FALLBACK] Trying add endpoint: ${fallbackUrl}`);
   642→
   643→        try {
   644→          const fallbackStart = Date.now();
   645→          const fallbackRes = await fetch(fallbackUrl, {
   646→            method: "POST",
   647→            headers: { "Content-Type": "application/json" },
   648→            body: JSON.stringify(fallbackPayload),
   649→            signal: AbortSignal.timeout(30000),
   650→          });
   651→          const fallbackDuration = Date.now() - fallbackStart;
   652→
   653→          const fallbackText = await fallbackRes.text();
   654→
   655→          console.log(`[Transfer][FALLBACK] HTTP ${fallbackRes.status}, duration: ${fallbackDuration}ms`);
   656→          console.log(`[Transfer][FALLBACK] Response: ${fallbackText.substring(0, 500)}`);
   657→
   658→          let fallbackData: Record<string, unknown>;
   659→          try {
   660→            fallbackData = JSON.parse(fallbackText);
   661→          } catch {
   662→            console.error(`[Transfer][FALLBACK] Non-JSON response`);
   663→            fallbackData = {};
   664→          }
   665→
   666→          // Use improved success check
   667→          const fallbackCheck = isTransferSuccess(fallbackRes.status, fallbackData);
   668→          if (fallbackCheck.success) {
   669→            let responseData = fallbackData.data;
   670→            if (typeof fallbackData.data === "string" && fallbackData.data) {
   671→              const decrypted = decryptAES(fallbackData.data);
   672→              if (decrypted) {
   673→                try { responseData = JSON.parse(decrypted); } catch { responseData = { rawDecrypted: decrypted }; }
   674→              }
   675→            }
   676→
   677→            console.log(`[Transfer][FALLBACK][SUCCESS] Team added via fallback (confidence: ${fallbackCheck.confidence})`);
   678→
   679→            return NextResponse.json({
   680→              status: "success",
   681→              method: "fallback_add",
   682→              endpoint: fallbackUrl,
   683→              platform: app,
   684→              operation: "add",
   685→              data: responseData || null,
   686→              httpStatus: fallbackRes.status,
   687→              confidence: fallbackCheck.confidence,
   688→              message: fallbackCheck.confidence === 'tentative'
   689→                ? `Team added to ${app} (verified by proxy, fallback)`
   690→                : `Team added to ${app} successfully (fallback from edit)`,
   691→            }, { status: 200 });
   692→          }
   693→
   694→          const fallbackError = (fallbackData?.message as string) || (fallbackData?.error as string) || "Unknown fallback error";
   695→          console.error(`[Transfer][FALLBACK][FAILED] ${fallbackError}`);
   696→
   697→          // Check for token expiry in fallback too
   698→          if (isConfirmedTokenExpiry(fallbackError, fallbackData)) {
   699→            return NextResponse.json({
   700→              status: "fail",
   701→              error: `Session expired on ${app}. Please re-link your account via OTP verification.`,
   702→              code: "TOKEN_EXPIRED",
   703→              needsReauth: true,
   704→              backendError: fallbackError,
   705→            }, { status: 200 });
   706→          }
   707→
   708→          if (fallbackError && fallbackError !== "Something Went Wrong!") {
   709→            lastError = fallbackError;
   710→          }
   711→          lastHttpStatus = fallbackRes.status;
   712→        } catch (fallbackErr) {
   713→          console.error(`[Transfer][FALLBACK][ERROR] ${fallbackErr instanceof Error ? fallbackErr.message : "unknown"}`);
   714→        }
   715→      }
   716→    }
   717→
   718→    // Return detailed error
   719→    console.error(`\n${"!".repeat(60)}`);
   720→    console.error(`[Transfer][FINAL] ALL ENDPOINTS FAILED for ${app} ${operation}`);
   721→    console.error(`[Transfer][FINAL] Backend error: ${lastError}`);
   722→    console.error(`[Transfer][FINAL] HTTP status: ${lastHttpStatus}`);
   723→    console.error(`[Transfer][FINAL] Endpoints tried: ${endpointChain.join(", ")}`);
   724→    console.error(`[Transfer][FINAL] Token: len=${preparedToken.length}, isJWT=${preparedToken.startsWith("eyJ")}, isJSON=${preparedToken.startsWith("{")}`);
   725→    console.error(`${"!".repeat(60)}\n`);
   726→
   727→    let displayError = lastError || "Unknown transfer error";
   728→
   729→    if (displayError === "All proxy attempts failed" || displayError.includes("proxy attempts failed")) {
   730→      displayError = `The ${app} server rejected the transfer. Your session may have expired. Please re-link your account via OTP verification and try again.`;
   731→    } else if (displayError === "Something Went Wrong!") {
   732→      displayError = `The ${app} server returned an error. Your session has likely expired - please re-link your account via OTP verification.`;
   733→    } else if (displayError === "Proxy returned 400") {
   734→      displayError = `Your ${app} session has expired (the platform rejected the request). Please re-link your account via OTP verification.`;
   735→    } else if (displayError === "Error while transfering the team!") {
   736→      displayError = `Failed to transfer team on ${app}. Your auth session has likely expired. Try re-linking your ${app} account via OTP verification.`;
   737→    } else if (displayError.includes("non-JSON response") && lastHttpStatus === 0) {
   738→      displayError = `Could not reach the ${app} transfer server. Please check your internet connection and try again.`;
   739→    }
   740→
   741→    return NextResponse.json({
   742→      status: "fail",
   743→      error: displayError,
   744→      code: "TRANSFER_ERROR",
   745→      needsReauth: false,
   746→      backendError: lastError,
   747→      httpStatus: lastHttpStatus,
   748→      platform: app,
   749→      operation: operation,
   750→      endpointsTried: endpointChain,
   751→      details: {
   752→        matchId: String(matchId),
   753→        captain: finalCaptain,
   754→        vice_captain: finalViceCaptain,
   755→        playerCount: numericPlayerIds.length,
   756→        teamId: id || null,
   757→        tokenLength: preparedToken.length,
   758→        tokenIsJWT: preparedToken.startsWith("eyJ"),
   759→        tokenIsJSON: preparedToken.startsWith("{"),
   760→      },
   761→    }, { status: 200 });
   762→
   763→  } catch (error) {
   764→    console.error("[Transfer][ROUTE_ERROR]", error);
   765→    return NextResponse.json({
   766→      status: "fail",
   767→      error: `Server error: ${error instanceof Error ? error.message : "Unknown error"}`,
   768→      code: "SERVER_ERROR",
   769→      needsReauth: false,
   770→    }, { status: 200 });
   771→  }
   772→}
