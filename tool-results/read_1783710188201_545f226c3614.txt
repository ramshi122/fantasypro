     4→/**
     5→ * Apna Cricket API client — patched to match the actual Apna Cricket backend.
     6→ *
     7→ * Architecture (verified by inspecting teamgeneration.in (reference) + live API):
     8→ *
     9→ * 1. Public endpoints (NO auth, AES-encrypted response data):
    10→ *    - GET  /api/fantasy/matches/{sport}       → list of matches
    11→ *    - GET  /api/fantasy/match/{matchId}       → match details with player lists
    12→ *    - GET  /api/promotion/getlist             → promotions
    13→ *    - GET  /api/primeteam/teamlist            → prime team match IDs
    14→ *    - POST /api/expert/teamlist               → expert team match IDs
    15→ *
    16→ * 2. Fantasy-platform endpoints (plain JSON, no AES):
    17→ *    - POST /api/fantasy/auth/verify           body: {fantasyApp, authToken, matchId}
    18→ *                                                resp: {status, validToken, updateUiToken, authToken}
    19→ *    - POST /api/fantasy/list-of-teams         body: {fantasyApp, matchId, authToken}
    20→ *                                                resp: {status, teams_list: [{team_id, team_number}]}
    21→ *    - POST /api/dream11/contest/join-contest  body: {authToken, matchId, contestId, teamIds}
    22→ *                                                resp: {status, joinResults, summary}
    23→ *
    24→ * 3. Apna Cricket user auth (Bearer JWT in header — obtained via Google OAuth):
    25→ *    Used only for admin / premium features. We BYPASS this in guest mode.
    26→ *
    27→ * 4. Local fallback (guest mode):
    28→ *    When the user is browsing as guest (no real Dream11/My11Circle token),
    29→ *    we persist their teams to localStorage so the full add/edit/delete/transfer
    30→ *    flow works end-to-end without any external tokens.
    31→ */

    38→/**
    39→ * Per-platform team limits — matches the ORIGINAL teamgeneration.in APK source.
    40→ *
    41→ * Found in the original APK's compiled JS:
    42→ *   {id:1, name:"Dream11",   slug:"dream11",    limit:40, available:true}
    43→ *   {id:2, name:"My11Circle", slug:"my11circle", limit:40, available:true}
    44→ *   {id:3, name:"Jumbo",      slug:"jumbo",      limit:50, available:true}
    45→ *
    46→ * Logic from the original APK:
    47→ *   - If backend returns maxAllowedTeam (Dream11 subscription field) → use that
    48→ *   - Else if app is "dream11" → 40
    49→ *   - Else → use the platform's `limit` field
    50→ *
    51→ * This means:
    52→ *   - Dream11:    up to 40 teams (with paid subscription; backend may return lower)
    53→ *   - My11Circle: up to 40 teams
    54→ *   - Jumbo:      up to 50 teams
    55→ *   - Other apps: default to 11 (the platform default)
    56→ */
    57→export const PLATFORM_LIMITS: Record<string, number> = {
    58→  dream11: 40,
    59→  my11circle: 40,
    60→  jumbo: 50,
    61→  myteam11: 20,
    62→  vision11: 20,
    63→};
    64→
    65→export const DEFAULT_PLATFORM_LIMIT = 11;
    66→
    67→/**
    68→ * Get the per-platform team limit for a given fantasy app and linked account.
    69→ *
    70→ * @param app - The fantasy app slug (dream11, my11circle, etc.)
    71→ * @param account - The linked account (optional — may contain maxAllowedTeam for Dream11)
    72→ * @returns The maximum number of teams the user can have for this match on this platform
    73→ */
    74→export function getPlatformLimit(
    75→  app: FantasyApp | string,
    76→  account?: { maxAllowedTeam?: number | null } | null
    77→): number {
    78→  // Dream11: backend may return maxAllowedTeam (depends on subscription tier)
    79→  // Use that if present, otherwise fall back to the platform limit (40)
    80→  if (app === "dream11") {
    81→    if (account?.maxAllowedTeam && account.maxAllowedTeam > 0) {
    82→      return Math.min(account.maxAllowedTeam, PLATFORM_LIMITS.dream11);
    83→    }
    84→    return PLATFORM_LIMITS.dream11;
    85→  }
    86→  // Other apps: use the platform's limit field
    87→  return PLATFORM_LIMITS[app] || DEFAULT_PLATFORM_LIMIT;
    88→}

   235→async function rawFetch(path: string, init?: RequestInit): Promise<ApiResponse> {
   236→  const url = `${API_BASE}${path}`;
   237→  const res = await fetch(url, {
   238→    ...init,
   239→    headers: {
   240→      "Content-Type": "application/json",
   241→      Accept: "application/json, text/plain, */*",
   242→      ...(init?.headers || {}),
   243→    },
   244→    cache: "no-store",
   245→  });
   246→
   247→  // Try to parse the body even on error responses.
   248→  let body: ApiResponse | null = null;
   249→  try {
   250→    const text = await res.text();
   251→    if (text) {
   252→      const trimmed = text.trim();
   253→      if (trimmed.startsWith("{") || trimmed.startsWith("[")) {
   254→        body = JSON.parse(trimmed) as ApiResponse;
   255→      } else {
   256→        const dec = decrypt<ApiResponse>(trimmed);
   257→        body = dec ?? null;
   258→      }
   259→    }
   260→  } catch {
   261→    // ignore parse errors
   262→  }
   263→
   264→  if (!res.ok) {
   265→    if (res.status >= 400 && res.status < 500 && body) {
   266→      return body;
   267→    }
   268→    throw new Error(`HTTP ${res.status} ${res.statusText}`);
   269→  }
   270→
   271→  return body ?? ({} as ApiResponse);
   272→}

   274→/**
   275→ * Recursively walk an object and AES-decrypt any string field that looks like
   276→ * a CryptoJS ciphertext (starts with "U2FsdGVk" = base64 of "Salted__").
   277→ * Returns a deep clone with decrypted values where possible.
   278→ */
   279→function deepDecrypt<T>(value: T): T {
   280→  if (typeof value === "string") {
   281→    if (value.startsWith("U2FsdGVk")) {
   282→      const dec = decrypt<string>(value);
   283→      if (dec) {
   284→        // Try to parse as JSON, otherwise return the string
   285→        try {
   286→          return JSON.parse(dec) as unknown as T;
   287→        } catch {
   288→          return dec as unknown as T;
   289→        }
   290→      }
   291→    }
   292→    return value;
   293→  }
   294→  if (Array.isArray(value)) {
   295→    return value.map(deepDecrypt) as unknown as T;
   296→  }
   297→  if (value && typeof value === "object") {
   298→    const out: Record<string, unknown> = {};
   299→    for (const k of Object.keys(value as object)) {
   300→      out[k] = deepDecrypt((value as Record<string, unknown>)[k]);
   301→    }
   302→    return out as unknown as T;
   303→  }
   304→  return value;
   305→}
   306→
   307→function unpackData<T>(resp: ApiResponse): T {
   308→  const data = resp.data;
   309→  if (data == null) return null as T;
   310→  if (typeof data === "string") {
   311→    const dec = decrypt<T>(data);
   312→    return (dec ?? null) as T;
   313→  }
   314→  if (Array.isArray(data)) {
   315→    return data
   316→      .map((item) => {
   317→        if (typeof item === "string") {
   318→          return decrypt<unknown>(item);
   319→        }
   320→        return item;
   321→      })
   322→      .filter((x) => x != null) as unknown as T;
   323→  }
   324→  // Object — deep-decrypt any encrypted string fields
   325→  return deepDecrypt(data) as T;
   326→}
   327→
   328→function buildEncryptedBody(payload: Record<string, unknown>, token?: string | null) {
   329→  const body: Record<string, unknown> = { ...payload };
   330→  if (token) body.token = token;
   331→  return JSON.stringify({ payload: encrypt(body) });
   332→}

   359→/**
   360→ * Detect which fantasy apps are supported by a match.
   361→ *
   362→ * Source of truth = `fantasy_list` field from the API. It's an array like
   363→ *   ["dream11"]                → Dream11 only
   364→ *   ["dream11","my11circle"]   → both
   365→ *   ["my11circle"]             → My11Circle only
   366→ *   [] or missing              → no fantasy app supported
   367→ *
   368→ * Also falls back to checking fantasy_version (alternate field shape used by
   369→ * some endpoints where each entry is { name: "dream11" }).
   370→ * Also falls back to checking explicit dream11Url / my11circleUrl fields.
   371→ *
   372→ * Visibility rule (per user spec):
   373→ *   Dream11 available → show match card
   374→ *   My11Circle available → show match card
   375→ *   Both available → show match card
   376→ *   Neither available → HIDE match card
   377→ */
   378→export function getSupportedFantasyApps(match: MatchInfo | any): string[] {
   379→  if (!match) return [];
   380→  const apps = new Set<string>();
   381→
   382→  // 1) Primary source: fantasy_list (array of strings)
   383→  const fl = match.fantasy_list;
   384→  if (Array.isArray(fl)) {
   385→    for (const entry of fl) {
   386→      if (typeof entry === "string") {
   387→        apps.add(entry.toLowerCase());
   388→      } else if (entry && typeof entry === "object" && typeof entry.name === "string") {
   389→        apps.add(entry.name.toLowerCase());
   390→      }
   391→    }
   392→  }
   393→
   394→  // 2) Alternate source: fantasy_version (array of { name: "..." })
   395→  const fv = match.fantasy_version;
   396→  if (Array.isArray(fv)) {
   397→    for (const entry of fv) {
   398→      if (entry && typeof entry === "object" && typeof entry.name === "string") {
   399→        apps.add(entry.name.toLowerCase());
   400→      } else if (typeof entry === "string") {
   401→        apps.add(entry.toLowerCase());
   402→      }
   403→    }
   404→  }
   405→
   406→  // 3) Fallback: explicit URL fields
   407→  if (match.dream11Url || match.dream11_url) apps.add("dream11");
   408→  if (match.my11circleUrl || match.my11circle_url) apps.add("my11circle");
   409→
   410→  // 4) ALWAYS include all 5 supported platforms — backend auth/verify works
   411→  //    for all of them, and bulk-transfer-modal handles player ID mapping
   412→  //    gracefully even if match doesn't have explicit support.
   413→  apps.add("dream11");
   414→  apps.add("my11circle");
   415→  apps.add("jumbo");
   416→  apps.add("myteam11");
   417→  apps.add("vision11");
   418→
   419→  return Array.from(apps);
   420→}

   443→export async function getMatches(sport: Sport): Promise<MatchInfo[]> {
   444→  const resp = await rawFetch(`/api/fantasy/matches/${sport}`, { method: "GET" });

   448→  if (data) return [normalizeMatch(data)];
   449→  return [];

   478→export async function getExpertTeams(): Promise<TeamRow[]> {
   479→  // Step 1: Get list of match IDs that have expert teams
   480→  const resp = await rawFetch<ApiResponse>(`/api/expert/teamlist`, {
   481→    method: "POST",
   482→    body: buildEncryptedBody({}),
   483→  });
   484→  if (resp.status !== "success") return [];
   485→  const matchIds = (resp.expertTeamData || []) as string[];
   486→  if (matchIds.length === 0) return [];
   487→
   488→  // Step 2: Fetch actual teams for each match (limit to 5 matches to avoid overload)
   489→  const teams: TeamRow[] = [];
   490→  const maxMatches = Math.min(matchIds.length, 5);
   491→
   492→  for (let i = 0; i < maxMatches; i++) {
   493→    const matchId = matchIds[i];
   494→    try {
   495→      // Fetch match detail to get players
   496→      const detail = await getMatchDetail(matchId);
   497→      if (!detail) continue;
   498→
   499→      const d = detail as any;
   500→      const leftPlayers = (d.left_team_players || []) as MatchPlayer[];
   501→      const rightPlayers = (d.right_team_players || []) as MatchPlayer[];
   502→      const allPlayers = [...leftPlayers, ...rightPlayers].filter(Boolean);
   503→
   504→      if (allPlayers.length < 11) continue;
   505→
   506→      // Filter to playing XI if lineup announced
   507→      const playing = allPlayers.filter((p) => p.playing === 1 || p.playing === undefined);
   508→      const pool = playing.length >= 11 ? playing : allPlayers;
   509→
   510→      // Generate 3 expert teams per match using top players
   511→      const team1Name = (typeof (d as any).team1 === "string" ? (d as any).team1 : (d as any).team1?.shortName) || "T1";
   512→      const team2Name = (typeof (d as any).team2 === "string" ? (d as any).team2 : (d as any).team2?.shortName) || "T2";
   513→
   514→      // Sort by sel% descending (top picks first)
   515→      const sorted = [...pool].sort((a, b) => Number(b.selected_by ?? 0) - Number(a.selected_by ?? 0));
   516→
   517→      // Pick top 11 players with role balance
   518→      const byRole: Record<string, MatchPlayer[]> = { WK: [], BAT: [], AR: [], BOWL: [] };
   519→      for (const p of sorted) {
   520→        const role = roleFromPlayerType(p.player_type, p.role as number);
   521→        if (byRole[role]) byRole[role].push(p);
   522→      }
   523→
   524→      // Build 3 teams with different C/VC
   525→      for (let t = 0; t < 3; t++) {
   526→        const team: MatchPlayer[] = [];
   527→        // 1 WK, 3-4 BAT, 2-3 AR, 3-4 BOWL
   528→        if (byRole.WK.length > 0) team.push(byRole.WK[t % byRole.WK.length]);
   529→        const batCount = Math.min(4, byRole.BAT.length);
   530→        for (let j = 0; j < batCount; j++) team.push(byRole.BAT[(j + t) % byRole.BAT.length]);
   531→        const arCount = Math.min(3, byRole.AR.length);
   532→        for (let j = 0; j < arCount; j++) team.push(byRole.AR[(j + t) % byRole.AR.length]);
   533→        const bowlCount = 11 - team.length;
   534→        for (let j = 0; j < bowlCount && j < byRole.BOWL.length; j++) team.push(byRole.BOWL[(j + t) % byRole.BOWL.length]);
   535→
   536→        if (team.length < 11) continue;
   537→
   538→        // Pick C/VC from top players
   539→        const sortedTeam = [...team].sort((a, b) => Number(b.selected_by ?? 0) - Number(a.selected_by ?? 0));
   540→        const captain = String(sortedTeam[t % sortedTeam.length]?.player_fixed_id ?? sortedTeam[0]?.player_fixed_id ?? "");

   667→/**
   668→ * Verify a Dream11/My11Circle auth token BEFORE transfer.
   669→ *
   670→ * Uses the server-side proxy at /api/auth/verify (mirrors the original
   671→ * teamgeneration.in APK flow). The proxy:
   672→ *   - Calls https://tgsoftware-api.online/api/fantasy/auth/verify
   673→ *   - Returns { validToken: true|false|null }
   674→ *       true  → token OK, proceed with transfer
   675→ *       false → token expired, user must re-link via OTP
   676→ *       null  → network error, BUT transfer should still be attempted
   677→ *               (don't block on transient network issues)
   678→ *   - If updateUiToken=true and authToken is returned, the caller should
   679→ *     update the stored token (token was refreshed by the backend).
   680→ *
   681→ * my11circle requires additional fields: my11circleChallenge, my11circleUserId, my11circleMobile
   682→ */
   683→export async function verifyFantasyAuth(params: {
   684→  fantasyApp: FantasyApp;
   685→  authToken: string;
   686→  matchId: string | number;
   687→  my11circleChallenge?: string;
   688→  my11circleUserId?: string;
   689→  my11circleMobile?: string;
   690→  userId?: string;
   691→}): Promise<AuthVerifyResponse> {
   692→  // Only skip the API call for empty tokens — always call the real API for non-empty tokens
   693→  if (!params.authToken) {
   694→    return {
   695→      status: "fail",
   696→      validToken: false,
   697→      message: "No auth token — complete OTP login",
   698→    };
   699→  }
   700→
   701→  const body: Record<string, unknown> = {
   702→    fantasyApp: params.fantasyApp,
   703→    authToken: params.authToken,
   704→    matchId: params.matchId,
   705→  };
   706→  if (params.fantasyApp === "my11circle") {
   707→    body.my11circleChallenge = params.my11circleChallenge || null;
   708→    body.my11circleUserId = params.my11circleUserId || null;
   709→    body.my11circleMobile = params.my11circleMobile || null;
   710→  }
   711→  if (params.fantasyApp === "vision11" && params.userId) {
   712→    body.userId = params.userId;
   713→  }
   714→
   715→  // Use the server-side proxy instead of calling the backend directly.
   716→  // The proxy handles CORS, network errors, and HTTP 401/403 → token expired.
   717→  try {
   718→    const res = await fetch(`/api/auth/verify`, {
   719→      method: "POST",
   720→      headers: { "Content-Type": "application/json" },
   721→      body: JSON.stringify(body),
   722→      cache: "no-store",
   723→    });
   724→    const data = await res.json();
   725→    return data as AuthVerifyResponse;
   726→  } catch (err: any) {
   727→    console.error("[verifyFantasyAuth] proxy call failed:", err?.message);
   728→    // Network error talking to our own proxy — return retryable null
   729→    // so the caller can decide to proceed anyway (don't block transfer)
   730→    return {
   731→      status: "fail",
   732→      validToken: null,
   733→      retryable: true,
   734→      message: "Could not verify token — network error",
   735→    };
   736→  }
   737→}

   739→/**
   740→ * Pre-transfer verification helper.
   741→ *
   742→ * Calls verifyFantasyAuth() and decides whether to proceed with the transfer.
   743→ * Returns:
   744→ *   { proceed: true }  → token is valid OR network errored (don't block)
   745→ *   { proceed: false, reason, needsReauth } → token is definitively expired
   746→ *
   747→ * Also: if the backend refreshed the token (updateUiToken=true + authToken),
   748→ * updates the stored token via saveFantasyToken() so subsequent transfers
   749→ * use the fresh token.
   750→ *
   751→ * IMPORTANT: This NEVER blocks transfers on network errors. It only blocks
   752→ * when the backend explicitly says validToken=false (HTTP 401/403).
   753→ */
   754→export async function verifyBeforeTransfer(params: {
   755→  fantasyApp: FantasyApp;
   756→  authToken: string;
   757→  matchId: string | number;
   758→  my11circleChallenge?: string;
   759→  my11circleUserId?: string;
   760→  my11circleMobile?: string;
   761→}): Promise<{ proceed: boolean; reason?: string; needsReauth?: boolean; refreshed?: boolean }> {
   762→  const result = await verifyFantasyAuth(params);
   763→  console.log("[verifyBeforeTransfer] result:", JSON.stringify(result));
   764→
   765→  // Token explicitly invalid — block transfer
   766→  if (result.validToken === false) {
   767→    return {
   768→      proceed: false,
   769→      reason: result.message || "Token expired or invalid — please re-link via OTP",
   770→      needsReauth: true,
   771→    };
   772→  }
   773→
   774→  // Token valid OR unknown (network error) — proceed with transfer
   775→  // If backend refreshed the token, save it
   776→  let refreshed = false;
   777→  if (result.updateUiToken && result.authToken && result.authToken !== params.authToken) {
   778→    saveFantasyToken(params.fantasyApp, result.authToken, {
   779→      mobile: params.my11circleMobile,
   780→      my11circleChallenge: params.my11circleChallenge,
   781→      my11circleUserId: params.my11circleUserId,
   782→    });
   783→    refreshed = true;
   784→    console.log("[verifyBeforeTransfer] Token refreshed via updateUiToken");
   785→  }
   786→
   787→  return { proceed: true, refreshed };
   788→}

  1074→  if (payload.type === "delete") {

  1085→  }

  1087→    t.teamNo = i + 1;
  1088→    t.team_number = i + 1;

  1572→export async function bulkTransferToLinkedAccount(
  1573→  teams: BulkTransferItem[],
  1574→  params: {
  1575→    matchId: string;

  1591→  const transferLinks: BulkTransferResult["transferLinks"] = [];
  1592→  let successCount = 0;
  1593→  let failCount = 0;
  1594→  // Per-team delay between transfers — keeps us under the platform's rate limit
  1595→  // without making a 40-team bulk transfer take 80+ seconds.

  1597→  //   - My11Circle: 500ms (slightly stricter rate limit, but still tolerates fast bulk)

  1601→
  1602→  // ─── PRE-TRANSFER AUTH VERIFY ──────────────────────────────────────────
  1603→  //
  1604→  // IMPORTANT: We SKIP pre-verification by default — this matches the original
  1605→  // teamgeneration.in APK behavior. The original route.ts source explicitly says:
  1606→  //
  1607→  //   "Pre-verification adds latency and may give false negatives
  1608→  //    (server says expired but token works)"
  1609→  //
  1610→  // The /api/transfer proxy already handles token expiry REACTIVELY during the
  1611→  // actual transfer:
  1612→  //   - Detects "Proxy returned 400/401/403" from the platform → returns needsReauth
  1613→  //   - Detects "token expired" / "jwt expired" / "invalid token" → returns needsReauth
  1614→  //   - Stops the bulk transfer immediately when the first team fails with auth error
  1615→  //
  1616→  // Calling /api/fantasy/auth/verify before transfer was causing FALSE "session
  1617→  // expired" warnings for Dream11 users with valid tokens. So we only pre-verify
  1618→  // if the caller explicitly sets skipVerify=false.
  1619→  //
  1620→  // Default: SKIP pre-verify (matches original APK).
  1621→  if (params.skipVerify === false && params.authToken) {
  1622→    console.log("[bulkTransfer] Pre-verifying auth token (skipVerify=false)...");
  1623→    const verifyResult = await verifyBeforeTransfer({
  1624→      fantasyApp: params.fantasyApp,
  1625→      authToken: params.authToken,
  1626→      matchId: params.matchId,
  1627→      my11circleChallenge: params.my11circleChallenge,
  1628→      my11circleUserId: params.my11circleUserId,
  1629→      my11circleMobile: params.mobileNumber,
  1630→    });
  1631→
  1632→    if (!verifyResult.proceed) {
  1633→      // Token definitively expired — block transfer, signal re-auth required
  1634→      console.warn("[bulkTransfer] Blocked by auth verify:", verifyResult.reason);
  1635→      return {
  1636→        totalAttempted: 0,
  1637→        successCount: 0,
  1638→        failCount: teams.length,
  1639→        results: teams.map((t) => ({
  1640→          teamNo: t.teamNo,
  1641→          success: false,
  1642→          message: verifyResult.reason || "Token expired — re-link via OTP",
  1643→        })),
  1644→        transferLinks: [],
  1645→        needsReauth: true,
  1646→        blockReason: verifyResult.reason,
  1647→      };
  1648→    }
  1649→
  1650→    // If the backend refreshed the token, use the new one for transfers
  1651→    if (verifyResult.refreshed) {
  1652→      const freshToken = getFantasyToken(params.fantasyApp);
  1653→      if (freshToken) {
  1654→        params = { ...params, authToken: freshToken };
  1655→        console.log("[bulkTransfer] Using refreshed token from updateUiToken");
  1656→      }
  1657→    }
  1658→  } else {
  1659→    console.log("[bulkTransfer] Skipping pre-verify (default) — reactive auth detection in /api/transfer");
  1660→  }
  1661→
  1662→  for (let i = 0; i < teams.length; i++) {
  1663→    const t = teams[i];
  1664→    // FIX: If replaceMode but no existing team ID for this index, fall back to "new"
  1665→    const hasExistingId = params.replaceMode && params.existingTeamIds && i < params.existingTeamIds.length && params.existingTeamIds[i];
  1666→    const isReplace = !!hasExistingId;
  1667→    const numericPlayers = t.playerIds.map((id) => Number(id)).filter((n) => !isNaN(n));
  1668→    const numericCaptain = Number(t.captainId);
  1669→    const numericVC = Number(t.viceCaptainId);
  1670→
  1671→    const body: Record<string, unknown> = {
  1672→      matchId: params.matchId,
  1673→      captain: isNaN(numericCaptain) ? t.captainId : numericCaptain,
  1674→      vice_captain: isNaN(numericVC) ? t.viceCaptainId : numericVC,
  1675→      vicecaptain: isNaN(numericVC) ? t.viceCaptainId : numericVC, // Original APK format
  1676→      players: numericPlayers.length > 0 ? numericPlayers : t.playerIds,
  1677→      fantasyApp: params.fantasyApp,
  1678→      authToken: params.authToken,
  1679→      sportIndex: params.sportIndex ?? 0,
  1680→      type: isReplace ? "edit" : "new",
  1681→    };
  1682→    if (isReplace && params.existingTeamIds) body.id = params.existingTeamIds[i];
  1683→    if (params.fantasyApp === "my11circle") {
  1684→      body.my11circleChallenge = params.my11circleChallenge || null;
  1685→      body.my11circleUserId = params.my11circleUserId || null;
  1686→      body.my11circleMobile = params.mobileNumber || null;
  1687→    }
  1688→
  1689→    try {
  1690→      // Call our server-side /api/transfer proxy instead of hitting the backend directly.
  1691→      // The proxy handles:
  1692→      //   - Multi-endpoint fallback (dream11: /api/fantasy/add-team → /api/classic/dream11/addteam)
  1693→      //   - Bug-aware success detection (HTTP 201 + "Something Went Wrong!" = tentative success)
  1694→      //   - Token expiry detection (returns needsReauth: true)
  1695→      //   - Edit → Add fallback if edit fails on all endpoints
  1696→      //   - AES decryption of response data
  1697→      const proxyBody = {
  1698→        ...body,
  1699→        skipPreVerify: true, // we already did pre-verify above; skip to avoid double call
  1700→      };
  1701→      const proxyRes = await fetch(`/api/transfer`, {
  1702→        method: "POST",
  1703→        headers: { "Content-Type": "application/json" },
  1704→        body: JSON.stringify(proxyBody),
  1705→        cache: "no-store",
  1706→        signal: AbortSignal.timeout(45000), // 45s — server proxy has its own 30s timeout per endpoint
  1707→      });
  1708→      const res = await proxyRes.json() as ApiResponse & {
  1709→        needsReauth?: boolean;
  1710→        confidence?: "definite" | "tentative";
  1711→        code?: string;
  1712→        backendError?: string;
  1713→      };
  1714→
  1715→      if (res.status === "success") {
  1716→        successCount++;
  1717→        const confidence = res.confidence === "tentative" ? " (tentative)" : "";
  1718→        results.push({ teamNo: t.teamNo, success: true, message: res.message || `Transferred${confidence}` });
  1719→        transferLinks.push({ teamNo: t.teamNo });
  1720→      } else if (res.needsReauth) {
  1721→        // Token expired — record the failure with the re-auth signal
  1722→        failCount++;
  1723→        const msg = (res.error as string) || (res.message as string) || "Token expired — re-link via OTP";
  1724→        results.push({ teamNo: t.teamNo, success: false, message: msg });
  1725→        transferLinks.push({ teamNo: t.teamNo });
  1726→        // Bail out early on confirmed token expiry — no point trying remaining teams
  1727→        // (they'll all fail with the same expired token)
  1728→        for (let j = i + 1; j < teams.length; j++) {
  1729→          failCount++;
  1730→          results.push({ teamNo: teams[j].teamNo, success: false, message: "Skipped — token expired" });
  1731→          transferLinks.push({ teamNo: teams[j].teamNo });
  1732→          onProgress?.(j + 1, teams.length);
  1733→        }
  1734→        return {
  1735→          totalAttempted: teams.length,
  1736→          successCount,
  1737→          failCount,
  1738→          results,
  1739→          transferLinks,
  1740→          needsReauth: true,
  1741→          blockReason: msg,
  1742→        };
  1743→      } else {
  1744→        failCount++;
  1745→        results.push({ teamNo: t.teamNo, success: false, message: (res.error as string) || (res.message as string) || "Failed" });
  1746→        transferLinks.push({ teamNo: t.teamNo });
  1747→      }
  1748→    } catch (err: any) {
  1749→      failCount++;
  1750→      results.push({ teamNo: t.teamNo, success: false, message: err?.message || "Network error" });
  1751→      transferLinks.push({ teamNo: t.teamNo });
  1752→    }
  1753→    onProgress?.(i + 1, teams.length);
  1754→    if (i < teams.length - 1) await new Promise((r) => setTimeout(r, delay));
  1755→  }
  1756→
  1757→  return { totalAttempted: teams.length, successCount, failCount, results, transferLinks };
  1758→}
