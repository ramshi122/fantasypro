// This script patches the Next.js standalone server.js to:
// 1. Load .env file before startup
// 2. PORT selection: PORT env takes priority, then FC_CUSTOM_LISTEN_PORT, then 3000
//    - When start.sh sets PORT=3000 (non-root), server uses 3000
//    - When start.sh unsets PORT (root, no Caddy), server uses FC_CUSTOM_LISTEN_PORT (81)
//    - Default fallback: 3000

const fs = require('fs');
const path = require('path');

const serverJsPath = path.join(__dirname, '..', '.next', 'standalone', 'server.js');

if (!fs.existsSync(serverJsPath)) {
  console.error('[patch-server] server.js not found at:', serverJsPath);
  process.exit(1);
}

let serverJs = fs.readFileSync(serverJsPath, 'utf8');

// Check if already patched
if (serverJs.includes('[PORT-FIX]')) {
  console.log('[patch-server] server.js already patched, skipping');
  process.exit(0);
}

// Patch 1: Insert .env loader before process.env.NODE_ENV = 'production'
if (!serverJs.includes('[ENV]')) {
  const envLoader = `// Load .env file - patch by scripts/patch-server.js
const _fs = require('fs'), _path = require('path');
try {
  const _envContent = _fs.readFileSync(_path.join(__dirname, '.env'), 'utf8');
  _envContent.split('\\n').forEach(_line => {
    const _idx = _line.indexOf('=');
    if (_idx > 0 && !_line.startsWith('#')) {
      const _key = _line.substring(0, _idx).trim();
      const _val = _line.substring(_idx + 1).trim();
      process.env[_key] = _val;
    }
  });
  console.log('[ENV] Loaded .env file');
} catch(_e) {
  console.log('[ENV] No .env file found, using defaults');
}
`;
  serverJs = serverJs.replace(
    "process.env.NODE_ENV = 'production'",
    envLoader + "process.env.NODE_ENV = 'production'"
  );
  console.log('[patch-server] Patched: .env loader inserted');
}

// Patch 2: PORT selection - PORT takes priority, then FC_CUSTOM_LISTEN_PORT, then 3000
// - When running as non-root: start.sh sets PORT=3000, unsets FC_CUSTOM_LISTEN_PORT
//   → server uses 3000, platform Caddy proxies :81 → :3000
// - When running as root with no Caddy: start.sh unsets PORT, lets FC_CUSTOM_LISTEN_PORT flow
//   → server uses FC_CUSTOM_LISTEN_PORT (81) directly for health checks
if (!serverJs.includes('[PORT-FIX]')) {
  serverJs = serverJs.replace(
    /const currentPort = parseInt\(process\.env\.PORT,\s*10\) \|\| 3000/,
    `// [PORT-FIX] PORT > FC_CUSTOM_LISTEN_PORT > 3000
// Non-root: start.sh sets PORT=3000 → server on :3000, Caddy proxies :81 → :3000
// Root (no Caddy): start.sh unsets PORT → server on FC_CUSTOM_LISTEN_PORT (81) directly
const _envPort = parseInt(process.env.PORT, 10)
const _fcPort = parseInt(process.env.FC_CUSTOM_LISTEN_PORT, 10)
const currentPort = (_envPort && _envPort > 0) ? _envPort : (_fcPort && _fcPort > 0) ? _fcPort : 3000`
  );
  console.log('[patch-server] Patched: PORT > FC_CUSTOM_LISTEN_PORT > 3000 priority');
}

fs.writeFileSync(serverJsPath, serverJs);
console.log('[patch-server] server.js fully patched with .env loader + smart PORT selection');
