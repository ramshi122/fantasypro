import { NextRequest, NextResponse } from "next/server";
import { firebaseGet, firebaseUpdate } from "@/lib/firebase";

// Admin secret for admin operations
const ADMIN_SECRET = "apna_cricket_admin_2024";

export async function POST(request: NextRequest) {
  try {
    const body = await request.json();
    const { accessKey, deviceFingerprint, action } = body;

    if (!accessKey || !deviceFingerprint) {
      return NextResponse.json(
        { success: false, message: "Access key and device fingerprint are required" },
        { status: 400 }
      );
    }

    const keyData = await firebaseGet(`/accessKeys/${accessKey}`) as {
      status?: string;
      deviceFingerprint?: string;
    } | null;

    if (!keyData) {
      return NextResponse.json(
        { success: false, message: "Invalid access key. Please check and try again." },
        { status: 404 }
      );
    }

    // Check if key is deactivated by admin
    if (keyData.status === "inactive" || keyData.status === "deleted") {
      return NextResponse.json(
        { success: false, message: "This access key has been deactivated. Contact admin for support." },
        { status: 403 }
      );
    }

    // Key exists and is active
    if (action === "activate") {
      // First-time activation
      if (!keyData.deviceFingerprint) {
        // Activate on this device
        await firebaseUpdate(`/accessKeys/${accessKey}`, {
          deviceFingerprint,
          activatedAt: new Date().toISOString(),
          status: "active",
        });

        return NextResponse.json({
          success: true,
          message: "Access key activated successfully!",
          keyStatus: "active",
        });
      } else if (keyData.deviceFingerprint === deviceFingerprint) {
        // Same device - already activated, allow access
        return NextResponse.json({
          success: true,
          message: "Access granted. Welcome back!",
          keyStatus: "active",
        });
      } else {
        // Different device - deny
        return NextResponse.json(
          {
            success: false,
            message: "This key is already activated on another device.",
          },
          { status: 403 }
        );
      }
    }

    if (action === "validate") {
      // Validate existing session
      if (keyData.deviceFingerprint === deviceFingerprint) {
        return NextResponse.json({
          success: true,
          message: "Session valid",
          keyStatus: keyData.status,
        });
      } else {
        return NextResponse.json(
          {
            success: false,
            message: "This key is already activated on another device.",
          },
          { status: 403 }
        );
      }
    }

    return NextResponse.json(
      { success: false, message: "Invalid action" },
      { status: 400 }
    );
  } catch (error) {
    console.error("Access key verification error:", error);
    return NextResponse.json(
      { success: false, message: "Server error. Please try again later." },
      { status: 500 }
    );
  }
}
