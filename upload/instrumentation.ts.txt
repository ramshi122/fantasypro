export async function register() {
  // Initialize database tables on server startup (Node.js runtime only)
  if (process.env.NEXT_RUNTIME === 'nodejs') {
    try {
      const dbModule = await import('@/lib/db')
      await dbModule.initDatabase()
      console.log('[Instrumentation] Database initialized successfully')
    } catch (error) {
      console.error('[Instrumentation] Database initialization failed (non-fatal):', error)
      // Don't throw - the app should still start even without database
    }
  }
}
